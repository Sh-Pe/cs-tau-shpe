	\subsection{השלמות מטופולוגיה}
	הרציונלים בני מנייה, כלומר נוכל למספר $\{q_n\}_{n = 1}^{\infty} = \Q$. עתה נתבונן בקבוצה הבאה: 
	\[ A = \bigcup_{n = 1}^{\infty}\cl{q_n - \frac{1}{2^{n}}, q_n + \frac{1}{2^{n}}} \]
	נבחין שהיא איחוד של קבוצות ולכן פתוחה. משום ש־$\Cl \Q = \R$, מתבקש לשאול האם $A = \R$. נוכל לענות על שאלה זו באמצעות מושג ה\textit{מידה} שנגדיר בהמשך, שיגדיר את גודל הקבוצה להיות סופי, ובפרט: 
	\[ m(A) \le \sum_{n = 1}^{\infty} \frac{1}{2^{n}} = 2 \]
	
	\textit{הערה: }משפט דיריכלה על קירובים דיופאנטים: 
	\[ \forall p, q \in \N\co \sof{\sqrt 2 - \frac{p}{q}} > \frac{1}{q^{2}} \]
	
	לכן, לדוגמה, ''אי אפשר לקרב את $\frac{1}{\sqrt 2}$ בצורה יחסית טובה ב־$A$``. 
	
	שימו לב שהקטעים בהגדרה של $A$ לא זרים, בהכרח, מצפיפות. עם זאת, ניתן להציג את $A$ כאיחוד קטעים זרים ופתוחים (היא לא קשירה). 
	\theo{\label{theo:1}תהא $A \subset \R$ קבוצה פתוחה. אז אפשר להציג את $A$ כאיחוד סופי או בן־מנייה של איחוד קטעים פתוחים זרים. }
	\defi{שקטע $I \subset A$ קטע פתוח מקסימלי ביחס ל־$A$ אם לא קיים $J \subset \R$ קטע פתוח כך ש־$I \subsetneq J \subset A$}
	נוכל להגדיר את המושג באופן דומה בעבור קבוצה פתוחה. 
	\rmark{לצורך המשפט, ''קטע פתוח`` במובן הרחב, כלומר בפרט קרנות $(\infty, a), (a, \infty)$ ו־$\R$. }
	\lem{תהא $A$ פתוחה. אז לכל $x_0 \in A$ קיים קטע מקסימלי שמכיל את $x_0$. }
	\begin{proof}
		נסמן: 
		\begin{gather*}
			\bg_* = \sup\ccb{\bg > x_0 \co [x_0, \bg) \subset A} \\
			\ag_* = \inf\ccb{\ag < x_0 \co (\ag, x_0] \subset A}
		\end{gather*}
		נבחין במספר הבחנות: 
		\begin{itemize}
			\item $\bm{\ag_* < x_0 < \bg_*}$\textbf{: }כי $x_0$ נקודת פנים ($x_0 \in \Int A$ כי $A$ פתוחה) לכן $x_0 \in (\ag, \bg) \subset A$ אז יש איזשהם $\ag, \bg$ לא טרוויאלים
			\item \textbf{הקטע }$\bm{(\ag_*, \bg_*) \subset A}$\textbf{: }ניקח $x \in (\ag_*, \bg_*)$. 
			\begin{itemize}
				\item אם $x > x_0$ אז קיים $\bg \ge x_0$ כך ש־$x \in [x_0, \bg) \subset A$
				\item אם $x \le x_0$ אז $\ag_* < x$ ולכן קיימת $\ag < x$ כך ש־$x \in [\ag, x_0) \subset A$
			\end{itemize}
			\item \textbf{הקטע $I \cod \bm{(\ag_*, \bg_*)}$ מקסימלי} (ביחס ל־$A$)\textbf{: }נניח בשלילה שלא היה מקסימלי, ואז היה קטע $(\ag, \bg)$ כך ש־$(\ag_*, \bg_*) \subsetneq (\ag, \bg) \subset A$, אך מהגדרת הסופרמום $\bg \le \bg_*$ ומהאינפימום $\ag \ge \ag_*$ כלומר $(\ag, \bg) =(\ag_*, \bg_*)$ בסתירה. 
		\end{itemize}
	\end{proof}
	נפנה להוכחת המשפט. 
	\begin{proof}[הוכחת משפט \ref{theo:1}]
		 נתונה קבוצה $A$. נביט באוסף הקטעים המקסימליים ביחס ל־$A$. 
		 \begin{itemize}
		 	\item \textbf{הקטעים הנ''ל פתוחים: }טרוויאלי, קטע פתוח הוא קבוצה פתוחה. 
		 	\item \textbf{איחודם מכסה את $\bm{A}$: }ישירות מטענת העזר, סביב כל נקודה יש קטע פתוח מקסימלי כזה. 
		 	\item \textbf{הקטעים הנ''ל זרים: }נניח ש־$I, J \subset A$ שני קטעים מקסימליים שנחתכים, ובשלילה $I \cap J \neq \vn$. אזי $I \cup J \subset A$ פתוח, ומכיל את הקטע המקסימלי $I$ כלומר $I = I \cup J$. באותו האופן $J = I \cup J = I$ ולכן הם שווים. 
		 	\item \textbf{האיחוד לכל היותר בן בנייה: }כי $\R$ הוא CII – בכל קטע ישנו רציונלי, הרציונלים שווים זה מזה (הם זרים) ומאקסיומת הבחירה הבת מנייה הגדרנו העתקה חח''ע לקבוצה בת מנייה. 
		 \end{itemize}\envendproof
	\end{proof}

	\subsection{מידת לבג ב־$\R^2$}
	
	כיצד נגדיר את מושג המידה? הרעיון הוא להתאים לקבוצה $A \subset \R^{d}$ מספר, שאמור לתאר את \textit{המידה} שלה. 
	\begin{itemize}
		\item כש־$d = 1$, \textit{מידה} משמעה אורך
		\item כש־$d = 2$, \textit{מידה} משמעה אורך
		\item כש־$d = 3$, \textit{מידה} משמעה נפח
	\end{itemize}
	
	נגדיר את מידת לבג במקרה הפרטי בו $d = 2$. הבנייה עובדת באותה המידה בכל הממדים. לאחר מכן נגדיר את המידה באופן כללי בכל מרחב מידה (כלומר קבוצה עם מבנה נוסף שמתאר אילו קבוצות מדידות ואילו לא). 
	
	תהא $A \subset \R^{d}$, ונרצה באיזשהו אופן להגדיר את $m(A)$ שמייצג את השטח שלה. אנחנו כבר יודעים לחשב שטח של מלבן $a \times b$ ששטחו $a \cdot b$. זו הנחת הייסוד שנעבוד איתה (ביחד עם אדטיביות של קבוצות זרות). 
	
	\defi{\textit{מלבן אלמנטרי} ב־$\R^{d}$ הוא קבוצה מהצורה $\prod_{i = 1}^{d}[a_i, b_i)$ כאשר $a_i \le b_i\in \R$. }
	
	נבחין שהגדרה זו בפרט כוללת את הקבוצה הריקה. נגדיר את השטח/מידה של מלבן אלמנטרי כזה להיות: 
	\[ m(A) \cod m_d(A) \cod \prod_{i = 1}^{d}(b_i - a_i) \]
	(מסמנים לעיתים ב־$m_d$ את המידה ל־$\R^{d}$). 
	 
	\begin{Lemma}[אדטיביות]
		נניח ש־$R, R_1 \dots R_N \subset \R^{d}$ קוביות אלמנטריות, כך ש־$R = \biguplus_{k = 1}^{N}R_k$ (איחוד זר). אזי: 
		\[ m(R) = \sum_{k = 1}^{N}m_2(R_k) \]\eqs
	\end{Lemma}
	\prooftrivial
	(הטריק שעושים כאן, הוא לקחת את כל הקוביות, ''להמשיך`` את הצלעות שלהן לישרים אינסופיים, להוכיח למקרה הזה ואז לעשות רדוקציה למקרה המקורי). 
	
	באותו האופן, נוכל להוכיח את הלמה הבאה: 
	\begin{Lemma}[תת־אדטיביות]
		יהיו $P, P_1 \dots P_N \subset \R^{d}$ קוביות אלמנטריות, כך ש־$P \subset \bigcup_{i = 1}^{N}P_i$, אזי $m(P) \le \sum_{i = 1}^{N}m(P_i)$. 
	\end{Lemma}
	
	לעיתים גם \textit{תת־אדטיביות} מכונה \textit{סאב־אדטיביות}. 
	
	נקבע מלבן אלמנטרי $\Og \subset \R^{d}$, ונעבוד בה''כ בתוכו לצורך הנוחות. 
	\defi{קבוצה $A \subset \Og$ נקראת \textit{קבוצה אלמנטרית} אם היא איחוד סופי של מלבנים אלמנטריים. }
	
	ההצגה לא יחידה (ניתן לייצגה כאיחוד סופי של מלבנים אלמנטריים שונים), אך מידת הקבוצה לא תלויה בהצגה. 
	
	\clm{ב־$\Og$, איחוד, חיתוך ומשלים של מספר סופי של קבוצות אלמנטריות, הוא קבוצה אלמנטרית. }
	\begin{proof}יהיו $P_1 \dots P_N, R_1 \dots R_T$ מלבנים אלמנטריים, כך ש־$B = \bigcup_{i = 1}^{T}R_i, \ A = \bigcup_{i = 1}^{N}P_i$ אלמנטריים. 
		\begin{itemize}
			\item איחוד: ברור ש־$A \cup B = \bigcup_{i = 1}^{T} R_i \cup \bigcup_{j = 1}^{N}P_i$ כנדרש. 
			\item חיתוך: בדומה לאיחוד, בשילוב עם העובדה שחיתוך מלבנים אלמנטריים נותנת מלבן אלמנטרי. 
			\item משלים: אם $\Og = \prod [a_i, b_i)$ ומלבן אלמנטרי $R = \prod [x_i, y_i) \subset \Og$ אז מדה־מורגן נקבל ש־: 
			\[ \ol R = \bigcup_{k = 1}^{d} \cl{\cl{[x_i, a_i) \cup [b_i, y_i)} \times \prod_{i \neq k} [x_i, y_i)}  \]
			איחוד של קוביות אלמנטריות ולכן קבוצה אלמנטרית. ואז ל־$A$ כללית נקבל ש־$\Og \setminus A = (\Og \setminus R_1) \cap \cdots \cap (\Og \setminus R_N)$. 
		\end{itemize}
	\end{proof}

	\nthlec{1}{2}
	
	\rmark{תוצאה של המשפט לעיל היא המקרה הפרטי לפיו $A \cup B, A \cap B, A \trio B, A^{c} = \Og \setminus A, A \setminus B$ כולן קבוצות אלמנטריות. מכאן ש־: 
	\begin{itemize}
		\item קבוצת הקבוצת האלמנטריות המוכלות ב־$\Og$ היא אלגברה בוליאנית. 
		\item כל $A \subset \Og$ אלמנטרית אפשר להציג כאיחוד \textit{זר} של מספר סופי של מלבנים אלמנטריים. 
	\end{itemize}}
	ההוכחה של המסקנה השנייה היא הוכחה דומה למה שכבר ראינו של להמשיך את המלבנים. 
	
	\textbf{תזכורת: }$A \trio B \cod (A \setminus B) \cup (B \setminus A)$
	
	\defi{אוסף תת־קבוצות של $\Og$ הסגור למשלים ולאיחוד סופיים, וכן מכיל את הקבוצה הריקה, נקרא \textit{אלגברה של קבוצות}. }
	\clm{\label{theo:boolring}בהינתן אלגברה של קבוצות מעל $\Og$, תחת ההגדרה: 
	\begin{align*}
		A + B \cod A \trio B && A \cdot B = A \cap B
	\end{align*}
	מקבלים חוג עם יחידה (שהוא $\vn$), שבו $\forall x \co x^{2} = x$. קבוצה שמקיימת תנאים אלו נקראת \textit{אלגברה בוליאנית}. }
	\defi{חוג שבו $\forall x \co x^{2} = x$ נקרא \textit{חוג בוליאני}. }
	\cola{חוג בוליאני היא חוג קומטטיבי ממאפיין $2$. }
	\rmark{יש איזומורפיזם בין קטגוריית החוגים הבוליאנים לבין קטגוריית האלגבראות הבוליאניות, שמנוסח במפורש במשפט \ref{theo:boolring}. לרוב משתמשים במושג אלגברה בוליאנית בהקשרים של תקב''צ, ובחוג בוליאני בהקשרים אלגבריים. }
	
	\defi{תהי $A \subset \R^{d}$ קבוצה אלמנטרית $A = \bigcup_{i = 1}^{N}P_i$ כאשר $P_i \subset \R^{d}$ מלבנים אלמנטריים. נגדיר את מידת $A$ להיות: 
	\[ m(A) \cod \sum_{i = 1}^{N}m(P_i) \]}
	\clm{\ 
	\begin{enumerate}
		\item $m(A)$ לא תלויה בהצגה של $A$ כאיחוד של מלבנים אלמנטרים זרים, 
		\item ההגדרה לעיל מזדהה עם ההגדרה הקודמת במקרה ש־$A$ מלבן אלמנטרי. 
	\end{enumerate}}
	ההוכחה של $1$ פשוטה מאדטיביות של מידת מלבנים, ו־$1$ גורר ישירות את $2$. 
	
	\clm{תכונות בסיסיות של מידת קבוצות אלמנטריות: 
	\begin{itemize}
		\item בהינתן $A, B \subset \Og$ אלמנטריות אז $A \cap B = \vn \implies m(A \cup B) = m(A) + m(B)$
		\item אם $A_1 \dots A_N$ אלמנטריות זרות בזוגות, אז $m\cl{\bigcup A_i} = \sum m(A_i)$ (באינדוקציה על $1$)
		\item אם $A_1 \dots A_N$ אלמנטריות, אז $m\cl{\bigcup A_i} \le \sum m(A_i)$ 
	\end{itemize}}
	
	ההוכחה של כולן מסתמכת על ''הרחבה`` של המלבנים. 
	
	\begin{Theorem}[תת־$\sg$־אדטיביות]\label{theo:r2subsigmaadd}
		נניח ש־$A, A_1 \dots \subset \R^{d}$ סדרת קבוצות אלמנטריות כך ש־$A \subset \bigcup_{n = 1}^{\infty} A_n$, אז: 
		\[ m(A) \le \sum_{n = 1}^{\infty}m(A_n) \]
	\end{Theorem} \
	\rmark{בהקשרים הללו, האות סיגמה לרוב מסמנת איחוד בן־מנייה. }
	התכונה לעיל קשורה לשלמות של $\R^{d}$ כמרחב מטרי. אם לדוגמה היינו עוברים ברציונליים, אז לדוגמה המלבן $I = [0, 1)^{d} \cap \Q^{d}$ היה איחוד של ''המון ישרים דקים`` וזה לא היה עובד. 
	
	הוכחת המשפט מסתמכת על רדוקציה מאיחוד בן־מנייה לאיחוד סופי (שם יש לנו בבירור סאב סיגמה אדטיביות מתכונות האלגברה), והדרך המתבקשת לעשות את זה היא באמצעות היינה־בורל. 
	\lem{תהא $A$ אלמנטרית ו־$\eg > 0$. אז קיימות אלמנטריות $A', \tl A\subset \R^{d}$ כך ש־:
	\begin{enumerate}[(A)]
		\item קירוב מבפנים ע''י קבוצה סגורה: $\Cl A' \subset A \ \land \ m(A \setminus A') < \eg$. 
		\item קירוב מבחוץ ע''י קבוצה פתוחה: $A \subset \Int(\tl A) \ \land \ m(\tl A \setminus A) < \eg$. 
	\end{enumerate}}\begin{proof}
		במלבנים, ההוכחה פשוטה (בונים מלבן קטן). במקרה של איחוד סופי של $n$ מלבנים, נקרב באופן זהה כל אחד מהמלבנים בעבור $ \frac{\eg}{n}$, ואז נאחד. פנים וסגור אינווריאנטים לאיחוד ולכן סיימנו. 
	\end{proof}
	''אנחנו לא פיזיקאים`` $\sim$ המרצה אחרי שהוא כתב את ההוכחה הזו. 
	
	\begin{proof}[הוכחת \ref{theo:r2subsigmaadd}]
		לאיחוד סופי – טרוויאלי. \textbf{הרעיון: }במקום להוכיח ש־$x \le y$, נוכיח ש־$\forall \eg \co x \le y + \eg$. נקבע $\eg > 0$ כלשהו. מטרתנו להוכיח ש־$m(A) \le \sum_{n = 1}^{\infty} m(A_n)$. נחליף את $A$ באלמנטרית $\Cl A' \subset A$ כך ש־$m(A \setminus A') \le \frac{\eg}{2}$, ובאופן דומה נחליף את $A_n$ באלמנטרית $\tl A_n$ כך ש־$A \subset \Int(\tl A_n)$ ו־$m(\tl A_n \setminus A_n) < \frac{\eg}{2^{n + 1}}$. נשים לב ש־$A'$ פרה־קומפקטית (היא חסומה + היינה־בורל) ולכן: 
		\[ \Cl A' \subset A \subset \bigcup_{n = 1}^{\infty}A_n \subset \bigcup_{n = 1}^{\infty}\Int(\tl A_n) \]
		הקבוצות הפתוחות $\Int(\tl A_1) \dots \Int(\tl A_n)$ כיסוי פתוח של הקומפט $\Cl A'$, ולכן ויש תת־כיסוי סופי $\ol{A'} \subset \bigcup_{r = 1}^{\infty}$. עתה, נשתמש בטענה לאיחוד סופי של אלמנטריות: 
		\[ \begin{WithArrows}
			m(A) &= m(A') + m(A \setminus A') \\
			&\le \frac{\eg}{2} + m(A') \Arrow[ll]{$A' \subset \bigcup_{r = 1}^{N}\tl A_n$}\\
			&\le \frac{\eg}{2} + \sum_{r = 1}^{N}m(\tl A_n) \\
			&\le \frac{\eg}{2} + \sum_{n = 1}^{\infty}\csb{m(A_n) + \frac{\eg}{2^{n + 1}}} \Arrow[ll]{$\sum_{n = 1}^{\infty}\frac{1}{2^{n + 1}} = \frac{1}{2}$}\\
			&= \eg + \sum_{n = 1}^{\infty}\cl{m(A_n)}
		\end{WithArrows} \]
	\end{proof}
	
	\cola{\label{col:elemntaryprop}אז מה אנחנו יודעים על קבוצות אלמנטריות ב־$\Og$? 
		\begin{enumerate}
			\item הן מהוות אלגברה (סגורות לאיחוד סופי ומשלים)
			\item המידה $m$ אדטיבית־סופית עבורן (כלומר $A \cap B = \vn \implies m(A \cup B) = m(A) + m(B)$)
			\item $m$ היא תת־$\sg$־אדטיבית (כלומר $A \subset \bigcup_{n = 1}^{\infty}A_n \implies m(A) \le \sum_{n = 1}^{\infty}m(A_n)$)
	\end{enumerate}}

	\subsubsection{מידה חיצונית}
	
	אלו שלושת התכונות \textit{היחידות} שאנחנו עומדים להשתמש בהן בהגדרת המידה לקבוצות שאינן אלמנטריות. הניסיון הראשון שלנו להגדיר מידה כזו תהיה מידה חיצונית. 
	
	\defi{\label{defi:outermeasure}תהא $A \subset \R^{d}$ קבוצה כלשהי. אז \textit{המידה החיצונית של $A$} היא: 
	\[ m^{*}(A) = \inf \ccb{\sum_{n = 1}^{\infty}m(P_n)\ \middle\vert \ A \subset \bigcup_{n = 1}^{\infty}P_n \ \mathrm{s.t.} \ \text{$_dA$ אלמנטריים}} \]\eqsw}
	הבחנה חשובה היא ש־$\Img m^{*} = [0, +\infty]$. יש קבוצות שהשטח עבורן ''אינסופי``. 
	
	\begin{itemize}
		\item \textbf{מה טוב? }אכן קבוצות שהיינו מצפים לשטח מסוים יצאו סבירות, לדוגמה השטח של מעגל היחידה הוא $\pi$. 
		\item \textbf{מה רע? }אם מאמינים באקסיומת הבחירה, ניתן לבנות $A, B \subset I^{d}$ \textit{זרות} כך ש־$m^{*}(A \cup B) < m^*(A) + m^*(B)$
	\end{itemize}
	
	הפתרון של הדבר הרע יהיה לצמצם את $m^*$ לתת קבוצה נחמדה של קבוצות ''מדידות``. 
	
	\rmark{את הרחבת מושג השטח, אפשר לבסס רק על שלוש התכונות במסקנה \ref{col:elemntaryprop}. מעשה את זה בפרק הבא. }
	\rmark{שימו לב שמידה אלמנטרית מוגדרת על כל $\R^{d}$ ולא רק על קוביה אלמנטרית $\Og \subset \R^{d}$. }
	
	
	\begin{Collary}[תכונות של $m^*$]
		\begin{enumerate}
			\item אם $A \subset \R^{d}$ ברת מנייה, אז $m^*(A) = 0$. למה? כי $A = \{x_i\}_{i = 1}^{\infty}$, ואז נוכל לכסות את $x_i$ על מלבן אלמנטרי ששטחו $\frac{\eg}{2^{i}}$, ולקבל $\forall \eg > 0 \co m^*(A) \le \eg$. 
			\item בהגדרה של $m^*(A)$, אפשר להחליף ''מלבנים אלמנטריים`` ב־''קבוצות אלמנטריות`` (כי איחוד בן נייה של קבוצות סופיות הוא בן מנייה). 
			\item בהגדרה של $m^*(A)$, אפשר להחליף את $A \subset \bigcup_{n = 1}^{\infty}A_n$ בתנאי $A \subset \bigcup_{n = 1}^{\infty}\ol{A_n}$ או בתנאי $A \subset \bigcup_{n= 1}^{\infty}\Int(A_n)$
			\item עבור קבוצה $A \subset \R^{d}$ אלמנטרית, מתקיים $m(A) = m^*(A)$. 
		\end{enumerate}
	\end{Collary}
	ההבחנה השנייה, תאפשר לנו בהמשך להגדיר מידה חיצונית נטו בעזרת מעין מידה אלמנטרית (מקיימת את שלושת התכונות) מעל אלגברה נחמדה מספיק. 
	
	\begin{proof}[הוכחה של מסקנה 4]
		ברור ש־$m^*(A) \le m(A)$ בעבור כיסוי עם איבר 1 – $A$ עצמה. יותר קשה להראות ש־$m^*(A) \ge m(A)$, כלומר צ.ל. בהינתן $A \subset \bigcup_{n = 1}^{\infty}A_n$ כאשר $A_n$ אלמנטריות, מתקיים $m(A) \le \sum_{n = 1}^{\infty} m(A_n)$. גם כאן התשובה פשוטה, מתת־$\sg$־אדטיביות. 
	\end{proof}
	
	''ביטול. אין הפסקה``. 
	
	\textbf{דוגמאות של מידה חיצונית. }
	\begin{enumerate}
		\item $m^*([0, 1)^{d}) = 1$ (כי זה אלמנטרי)
		\item $m^*(I^{d}) = 1$. חסם תחתון ברור מהסעיף הקודם. החסם העליון כי עבור $\eg > 0$ קל להראות ש־$m^*(A) \le 1 + \eg$ ע''י הכיסוי $[0, 1) \cup P_1 \cup P_2 \cup P_3 \cup P_4$ כאשר $P_1 \dots P_4$ מלבנים אלמנטריים סביב צלעות המלבן ממידה $\frac{\eg}{4}$ (העובי שלהם קטן כרצוננו). 
		\item באופן דומה להוכחה של $P_1 \dots P_4$, ברור ש־$m^*(I) = 0$ בתוך $\R^{d}$. 
	\end{enumerate}
	
	\theo{מידה חיצונית היא תת־$\sg$־אדטיבית, כלומר: 
	\[ m^*\!\Bigg(\smash{\overbrace{\bigcup_{n =1}^{\infty}A_n}^{A}}\Bigg) \!\le \sum_{n = 1}^{\infty}m^*(A_n) \]\eqsw}\begin{proof}
		ניקח $\eg > 0$ ונוכיח את הא''ש עד לכדי אפסילון. לכל $n$, קיימות אלמנטריות $P_{n, 1} \dots P_{n, 3} \dots \subset \R^{d}$ כך ש־$A_n \subset \bigcup_{k =1}^{\infty}P_{n, k}$,		וכמעט מתקבל ה־$\inf$, כלומר: 
		\[ \begin{WithArrows}
			\sum_{k = 1}^{\infty}m^*(P_{n, k}) &< m^*(A_n) + \frac{\eg}{2^{n}} \Arrow[up]{$A \subset \bigcup_{n, k}P_{n, k}$}\\
			m^*(A) &\le \sum_{n, k = 1}^{\infty}m^*(P_{n, k}) \le \sum_{n = 1}^{\infty}m^*(A_n) + \frac{\eg}{2^{n}} = \eg + \sum_{n = 1}^{\infty}m^*(A_n)
		\end{WithArrows} \]
		כנדרש. 
	\end{proof}
	
	\subsubsection{קבוצות מדידות}
	
	בעוד כמה שבועות, נראה $A, B$ (''כלל וכלל לא אלמנטריות``) זרות כך ש־$m^*(A \cup B) < m^*(A) + m^*(B)$. זו הבעיה העיקרית במידה חיצונית – זה אמור להיות אדטיבי ביחד לשתי קבוצות זרות. 
	
	הפתרון יהיה להצטמצם לקבוצה בשם קבוצת הקבוצות המדידות, שעליהן $m^*$ אדטיבית (מידת איחוד זר היא סכום המידות) ''רוב הקבוצות הן מדידות``, בהיבט הזה שחייבים להשתמש באקסיומת הבחירה כדי לבנות קבוצות לא מדידות. 
	
	לעת עתה, אנחנו עדיין עובדים בתחום תחום קבוע $\Og \subset \R^{d}$ כלשהו. 
	
	\defi{קבוצה $A \subset \Og$ נקראת \textit{מדידה} אם ניתן לקרבה ע''י אלמנטריות, כלומר לכל $\eg > 0$ קיימת אלמנטרית $P_\eg \subset \Og$ כך ש־$m^*(A \trio P_\eg) < \eg$. }
	\rmark{בלעז, measurable. }
	\rmark{יש גישה אלטרנטיבית, שקולה, בה מגדירים את $A$ להיות מדידה אמ''מ לכל $B$ מתקיים ש־$m^*(A) = m^*(A \cup B) + m^*(A \setminus B)$. אנחנו נעבוד עם הגישה המוגדרת לעיל, שהיא הגישה של לבג. }
	''נוכיח שזה שקול?`` ''יום אחד``
	
	\textbf{דוגמאות: }\begin{enumerate}
		\item כל אלמנטרית היא מדידה, ניתנת לקירוב ע''י עצמה. 
		\item עיגולים הם גם סבבה: עם בנייה מגעילה מספיק אפשר לעשות את זה. ניתן להוכיח בצורה יותר אלגנטית באמצעות קריטריון דרבו לאינטגרביליות רימן. נראה בהמשך שכל פתוחה/סגורה היא מדידה. 
		\item אם $A \subset \Og$ קב' בעלת מידה חיצונית $0$ היא בהכרח מדידה (ניתן לקרבה עם הקבוצה הריקה). 
	\end{enumerate}
	
	בתרגיל, נוכיח ש־$d(A, B) = m^*(A \trio B)$ מקיים את א''ש המשולש וזו פסאודומטריקה. יתרה מכך, הפונקציה $A \to m^*(A)$ רציפה (מליפשיציות). 
	
	\clm{אוסף הקבוצות המדידות ב־$\Og$ סגור למשלים ולאיחוד (מהווה אלגברה). }\begin{proof}
		יהיו $A, B \subset \Og$ מדידות, ונוכיח ש־$A \cup B$ מדיד. יהי $\eg >0$. רוצים למצוא $\eg$־קירוב אלמנטרי ל־$A \cup B$. ישנן אלמנטריות $P_\eg, Q_\eg$ שמקרבות את $A, B$ בהתאמה ברזולוציה של $\frac{\eg}{2}$ (כלומר $m^*(A \trio P_\eg) < \frac{\eg}{2}$ ו־$m^*(B \trio P_\eg) < \frac{\eg}{2}$). 
		
		נרצה לקרב את $A \cup B$ באמצעות $P_\eg \cup Q_\eg$. ואכן: 
		\[ (A \cup B) \trio (P_\eg \cup Q_\eg) \subset (A \trio P_\eg) \cup (B \trio Q_\eg) \]
		ומתת־אדטיביות של $m^*$ קיבלנו ש־$m^*((A \cup B) \trio (P_\eg \cup Q_\eg)) < \frac{\eg}{2} + \frac{\eg}{2} = \eg$ וסיימנו. 
		
		מה לגבי $\Og \setminus A$? יש אפסילון־קירוב ל־$A$ ע''י $P_\eg$ כך ש־$m^*(A \trio P_\eg) < \eg$. אך $A^c \trio P_\eg^c = A \trio P_\eg$ כלומר $m^*(A^c \trio P_\eg^c) = \eg$ כנדרש. 
	\end{proof}
	
	\clm{נניח שיש $A, B \subset \Og$ \textit{מדידות} וזרות. אז $m^*(A \cup B) = m^*(A) + m^*(B)$. }\begin{proof}
		א''ש אחד ידוע מתת־אדטיביות של $m^*$. הכיוון השני (צ.ל. $m^*(A \cup B) \le m^*(A) + m^*(B)$) לאו דווקא ברור. 
		
		כדי להוכיח את הא''ש השני, ניקח $\eg > 0$ ונסתפק בלהוכיח ש־$m^*(A \cup B) \ge m^*(A) + m^*(B) - 10\eg$. ממדידות $A, B$ קיימות אלמנטריות $P_\eg, Q_\eg \subset \Og$ כך ש־$m^*(A \trio P_\eg), m^*(B \trio Q_\eg) < \eg$. 
		
		[שימו לב שלמרות ש־$A \cap B = \vn$, ייתכן ש $P_\eg \cap Q_\eg \neq \vn$, אבל אפשר להראות שהחיתוך קטן]
		
		נבחין ש־:
		\[ P_\eg \cap Q_\eg \subset (P_\eg \setminus A) \cup (Q \eg \setminus B) \subset (P_\eg \trio A) \cup (Q_\eg \trio B) \]
		לכן $m^*(P_\eg \cap Q_\eg) \le m^*(P_\eg \trio A) + m^*(Q_\eg \trio B) < 2\eg$. מכיוון ש־$P_\eg, Q_\eg \subset \Og$ אלמנטריות, אז $m^*(P_\eg) = m(P_\eg)$ וכנ''ל לגבי $Q_\eg$ (כאשר $m$ המידה שהגדרנו רק לאלמנטריות, והוכחנו בה אדטיביות סופית) ולכן: 
		\[ m^*(P_\eg \cup Q_\eg) = m^*(P_\eg) + m^*(Q_\eg) - m^*(Q_\eg \cap P_\eg) \]
		עבור $x, y \in \R$ נסמן $x \approx y$ אם $\sof{x - y} < \eg$. לעיל הסקנו ש־$\smash{m^*(P_\eg \cup Q_\eg) \overset{2\eg}{\approx} m^*(P_\eg) + m^*(Q_\eg)}$. 
		
		בנוסף, ברור ש־$m^*(A) \approx m^*(P_\eg)$ ו־$m^*(B) \approx m^*(Q_\eg)$. 
		
		לגבי $A \cup B$, ראינו בהוכחה הקודמת ש־$m^*\csb{(A \cup B) \trio (P_\eg \cup Q_\eg)} \le m^*(A \trio P_\eg) + m^*(B \trio P_\eg) \le 2\eg$. לכן $m^*(A \cup B) \overset{2\eg}{\approx} m^*(P_\eg \cup Q_\eg)$. נציב למעלה ונקבל: 
		\[ m^*(A \cup B) \overset{2\eg + 2\eg + \eg + \eg}{\approx} m^*(A) + m^*(B) \]
		ובפרט עד לכדי $10\eg$ כנדרש. 
	\end{proof}
	
	\cola{\begin{enumerate}
			\item אפשר להמשיך באינדוקציה על $A_1 \dots A_n$ מדידות וזרות בזוגות ולקבל $m^*(\bigcup A_i) = \sum m^*(A_i)$. 
			\item עבור $A, B \subset \Og$ מדידות, נוכל לקבל הכלה והפרדה: 
			\[ m^*(A \cup B) = m^*(A) + m^*(B) - m^*(A \cap B) \]
			(כי $m^*(B) = m^*(A \cap B) + m^*(B \setminus A)$ ובאופן דומה על $A$)
	\end{enumerate}}
	
	''המשפט הבא הוא פלא``. ''במידה מסוימת, הקורס הזה הוא הערות למשפט הזה``
	\begin{Theorem}[סיגמה־אדטיביות של המידה]
		תהיינה $A_1, A_2 \dots \subset \Og$ סדרה בת־מנייה של קבוצות מדידות. אז: 
		\begin{enumerate}
			\item איחוד הבן מנייה $\bigcup_{n = 1}^{\infty}A_n$ של קבוצות מדידות הוא מדיד
			\item אם הקבוצות זרות בזוגות, אז $m^*\cl{\bigcup_{n = 1}^{\infty}A_n} = \sum_{n = 1}^{\infty}m^*(A_n)$
		\end{enumerate}
	\end{Theorem}
	\begin{proof}
		\begin{itemize}
			\item להוכחת החלק הראשון של המשפט, אפשר להניח בה''כ שהן זרות בזוגות, שכן אפשר תמיד להחליפן ב־: 
			\[ \begin{cases}
				A_1' = A_1 \\
				A_n' = A_n \setminus \cl{\bigcup_{i = 1}^{n - 1} A_i}
			\end{cases} \]
			קיבלנו קבוצות $A_1' \dots A_n'$ זרות, מדידות (קבוצת המדידות היא אלגברה, ו־$A_n'$ מתקבלת ע''י מספר סופי של חיתוכים ואיחודים), ואיחודן אותו הדבר ($\bigcup A_n = \bigcup A_n'$). 
			\item לכל $N$ סופי, ידוע ש־: 
			\[ \sum_{n= 1}^{N}m^*(A_n) = m^*\cl{\bigcup^{N}_{n = 1} A_n} \! \le m^*(A) < \infty \]
			מתת־$\sg$־אדטיביות. לכן הטור $\sum^{\infty}_{n  =1}m^*(A_n)$ מתכנס, כי סדרת הסכומים החלקיים חסומה (הרגע הוכחנו) ומונוטונית עולה (המידה חיובית). לכן הזנב קטן כרצוננו, כלומר לכל $\eg > 0$ ישנו $N_\eg$ כך ש־$\sum_{n = N_\eg}^{\infty}m^*(A_n) < \eg$ (זנב כל טור מתכנס). 
			
			\item למה $A = \bigcup_{n = 1}^{\infty}$ מדידה? יהי $\eg > 0$, נמצא קירוב אלמנטרי מתאים. הקבוצה $\bigcup_{n= 1}^{N_\eg}A_n$ מדידה, ולכן יש אלמנטרית $P_\eg$ שמקרבת אותה $m^*(P_\eg \trio \bigcup^{N_\eg}_{n = 1}A_n) < \eg$. נראה ש־$m^*(P_\eg \trio A) < 2\eg$. אכן: 
			\[ \begin{WithArrows}
				m^*(P_\eg \trio A) &= m^*\!\cl{P_\eg \trio \csb{\bigcup_{n = 1}^{N_\eg} \cup\  \bigcup_{\mathclap{n = N_\eg + 1}}^{\infty} A_n}} \Arrow[up]{סאב־אדטיביות} \\
				&= m^*\!\cl{P_\eg \trio \bigcup_{n = 1}^{n_\eg}} + m^*\!\cl{\bigcup_{n = N_\eg + 1}^{\inft}A_n} \\
				&= \eg + \eg = 2 \eg
			\end{WithArrows} \]
			בכך הוכחנו את חלק א' של המשפט. 
			\item עתה, נותר להראות ש־$m^*\cl{\bigcup_{n = 1}^{\infty}} = \sum_{n = 1}^{\infty}m^*(A_n)$ כאשר $A_n$ זרות. כיוון אחד יש לנו מתת־$\sg$־אדטיביות. 
			
			לגבי הכיוון השני, ממונוטוניות. לכל $N$ מתקיים: 
			\[ m^*\!\cl{\bigcup_{n = 1}^{\infty}A_n} \ge m^*\!\cl{\bigcup_{n = 1}^{N}A_n} = \sum_{n = 1}^{N}m^*(A_n) \]
			ולכן מסנדוויץ' סיימנו. 
		\end{itemize}\envendproof
	\end{proof}

	\defi{אוסף תת־קבוצות של $\Og$ הסגור למשלים ולאיחודים בני מנייה, וכן מכיל את הקבוצה הריקה, נקרא $\sg$־אלגברה (קרי: סיגמא אלגברה). }
	
	או במילים אחרות: אלגברה שעובדת גם לאיחודים בני מנייה (ולא רק סופיים). 
	
	יש סיגמה־אלגבראות מעוצמת הרצף, אך אין סיגמה־אלגברה (לא סופית) מעוצמה בת־מנייה. 
	
	לסיכום, הקבוצות המדידות ב־$\Og$ הוא $\sg$־אלגברה, והמידה החיצונית היא $\sg$־אדטיבית (כלומר היא אינווריאנטית לאיחוד קבוצות זרות). 
	
	\defi{עבור $A \subset \Og$ מדידה, מסמנים ב־$m(A) = m^*(A)$, וזו נקראת \textit{מידת לבג} של $A$. }
	
	זה כמובן מרחיב את ההגדרה הקודמת לקבוצות אלמנטריות. זה לא תלוי ב־$\Og$ (כי מידה חיצונית לא תלויה ב־$\Og$). 
	
	\textbf{דוגמאות: }
	\begin{enumerate}
		\item אם $A_1, A_2 \dots \subset \Og$ מדידות, אז גם $\bigcap_{n = 1}^{\infty}A_n$ מדיד (כי יש שימור של איחוד אינסופי ושל משלים, ומכללי דה מורגן, כלומר $\bigcap A_n = \cl{\bigcup A_n^{c}}^{c}$). 
		\item איחוד בן־מנייה של מלבנים אלמנטריים (או איחוד כל המלבנים הרציונליים שמוכלים בקבוצה מדידה). 
		\item כל קבוצה פתוחה $A \subset \Og$ מדידה. 
		\item כל קבוצה סגורה $A \subset \Og$ מדידה, זאת כי: 
		\[ \Og \setminus A = \underbrace{\Int \Og \setminus A}_{\ \!\text{פתוחה}} \cup \ B \]
		כאשר $B \subset \partial \Og$, ו־$\partial \Og$ זניחה וממונוטוניות וסגירות לאיחוד סיימנו. 
	\end{enumerate}

	\subsubsection{הגדרת מידה מחוץ ל־$\Og$}
	כרגע, קובעים מלבן אלמנטרי $\Og \subset \R^{d}$ ועובדים בתוכו. נרצה להתרחב לכל $\R^{d}$. 
	
	מומלץ לקרוא על המאמר של Falconer מ־1986. 
	
	תופעה מעניינת: אין קבוצה מדידה שמכילה ''חצי מהממשיים בכל סקאלה``. הכוונה היא שלא קיימת $A \subset \R$ מדידה כך שלכל $I \subset \R$ מתקיים $m(A \cup I) = \frac{m(I)}{2}$. 
	
	כשנרצה להתרחב מחוץ ל־$\Og$, נרצה להגדיר גם מידה לקבוצות לא חסומות. 
	
	\textbf{הרעיון: }נרצף את $\R^{d}$ עם מלבנים אלמנטריים, לדוגמה באמצעות $[n, n +1) \times [m, m + 1)$ לכל $n, m \in \Z$. נסמן את הריבועים בריצוף ב־$\Og_1, \Og_2 \dots $ ומן הסתם מתקיים ש־$\R^{d} = \biguplus^{\infty}_{i = 1} \Og_i$. 
	
	\defi{\label{defi:mesuareR2}קבוצה $A \subset \R^{d}$ תקרא \textit{מדידה לבג} אם $\forall n \co A \cap \Og_n$ מדידה ב־$\Og_n$, ונגדיר את \textit{מידת לבג של $A$} להיות: 
	\[ m(A) \cod \bigcup_{n = 1}^{\infty}m(A \cap \Og_n) \]\eqsw}
	
	\exe{נניח ש־$\Og_1, \Og_2$ מלבנים אלמנטריים, ונניח ש־$A \subset \Og_1 \cap \Og_2$. אז אם $A$ מדידה ביחס ל־$\Og_1$ היא גם מדידה ביחס ל־$\Og_2$. }
	\cola{הגדרת המדידות לא תלויה בבחירת הריצוף. }
	
	\clm{ההגדרה ב־\ref{defi:mesuareR2} מקיימת את התנאים הבאים: 
	\begin{enumerate}
		\item אוסף הקבוצות ב־$\R^{d}$ הוא $\sg$־אלגברה. 
		\item אם $A_1, A_2 \dots \subset \R^{d}$ מדידות וזרות, אז: 
		\[ m\cl{\bigcup_{n = 1}^{\infty}A_n\!} = \sum_{n = 1}^{\infty}m(A_n) \]
	\end{enumerate}\eqsw}
	\begin{proof}
		\begin{enumerate}
			\item ברור ש־$\vn$ מדידה (כי $\vn \cap \Og_n = \vn$ מדידה ב־$\Og_n$). סגירות למשלים ברורה כי: 
			\[ \R^{d} \setminus A = \biguplus{\Og _n \setminus A} \]
			ו־$\Og_n \setminus A$ מדידה לכל $n$ (כי שווה ל־$\Og_n\setminus (\Og_n \cap A)$). ישנה סגירות לאיחוד בן־מנייה: 
			\[ \cl{\bigcup_{i = 1}^{\infty}A_i} \cap \Og_n = \bigcup_{i= 1}^{\infty}\underbrace{\cl{A_i \cap \Og _n}}_{\text{מדידה ב־$_n\Og$}} \]
			\item יהיו $A_1, A_2 \dots $ כמו קודם, ואז: 
			\[ \begin{WithArrows}
				&m\cl{\bigcup_{i =1}^{\infty}A_i} \Arrow[up]{מההגדרה}\\
				= &\sum_{n = 1}^{\infty}m\cl{\cl{\bigcup_{i =1}^{\infty}A_i} \cap \Og_n}  \\
				 = & \sum_{n = 1}^{\infty}m\cl{\bigcup_{i = 1}^{\infty}\cl{A_i \cap \Og_n}} \Arrow[]{$\sg$־אדטיביות ב־$_n\Og$} \\
				 = &\sum_{n = 1}^{\infty}\sum_{i= 1}^{\infty}m\cl{A_i \cap \Og_n} \Arrow[]{החלפת סדר סכימה} \\
				 = & \sum_{i= 1}^{\infty}\cl{\sum_{n = 1}^{\infty}m\cl{A_i \cap \Og_n}}
			\end{WithArrows} \]
		\end{enumerate}
	\end{proof}
	
	\clm{לכל $A \subset \R^{d}$ מדידה לבג, מתקיים $m(A) = m^*(A)$}
	\rmark{זכרו שמידה חיצונית מוגדרת לכל $\R^{d}$. ראה \ref{defi:outermeasure}. }
	\begin{proof}
		למה $m(A) \ge m^*(A)$? כי: 
		\[ m(A) = \sum_{n =1}^{\infty}m(A \cap \Og _n) = \sum_{n = 1}^{\infty}m^*(A \cap \Og_n) \ge m^*\cl{\bigcup_{n = 1}^{\infty}\cl{A \cap \Og _n}} = m^*(A) \]
		למה $m(A) \le m^*(A)$? מספיק להראות את השוויון $m^*(A) \ge \sum_{n = 1}^{N}m(A \cap \Og _n)$ לכל $N$, ואז לקחת גבול וסיימנו. נקבע מלבן אלמנטרי שמכיל את $\Og_1 \dots \Og_n$ (יש מספר סופי, ולכן יש מלבן אלמנטרי שמכיל את כולם). הקבוצה $A \cap \Og_i$ מדידה ב־$\Og$ לכל $i$ (תרגיל למעלה), לכן $A \cap \Og = \bigcup_{i = 1}^{N}\cl{A \cap \Og_i}$ מדידה ב־$\Og$, ו־: 
		\[ m^*(A) \ge m^*(A \cap \Og) m(A \cap \Og) \ge \sum_{i =1}^{N}m(A \cap \Og_i) \]
		כאשר הא''שים לעיל נכונים ממונוטוניות, מדידות ותת־אדטיביות. 
		והוכחנו את הנדרש. 
	\end{proof}
	
	בפרט, הגדרת מידת לבג לא תלויה בריצוף. 
	
	\textbf{שאלה: }נתונות $A, B \subset \R^{d}$ מדידות. האם $m(A \cup B) = m(A) + m(B)- m(A \cap B)$? 
	
	\textbf{תשובה: }זה לאו דווקא מוגדר היטב. ייתכן שהקבוצות אין־סופיות, ואי אפשר לחסר אינסוף מאינסוף. מה שאפשר להגיד הוא ש־$m(A \cup B) + m(A \cap B) = m(A) + m(B)$. 
	
	\textbf{שאלה: }איזה קבוצות מדידות יש ב־$\R^{d}$? הוכחנו שיש: 
	\begin{itemize}
		\item אלמנטריות
		\item פתוחות (כאיחוד בן מנייה של אלמנטריות) וסגורות
		\item קבוצות ממידה (חיצונית) אפס
		\item ועוד
	\end{itemize}
	
	שאלה מתבקשת היא כמה קבוצות מדידות יש ב־$\R^{d}$. 
	\clm{יש $2^{\con}$ קבוצות מדידות ב־$\R^{d}$. }\begin{proof}
		החסם העליון פשוט, כי $\sof{\ps(\R^{d})} = 2^{\con}$. נדרש למצוא רק חסם תחתון. נחפש קבוצה מעוצמת הרצף, שכל תת־הקבוצות שלה מדידות. מספיק למצוא קבוצה ממידה אפס, שעוצמתה רצף. ב־$d \ge 2$ ההוכחה פשוטה: ניקח פשוט קו ישר. ב־$d = 1$ נוכל להשתמש בקבוצת קנטור. 
	\end{proof}
	
	נתבונן בקבוצה החביבה הבאה: 
	\[ A_\eg = \bigcup_{n = 1}^{\infty}\cl{q_n - \frac{\eg}{2^{n}}, q + \frac{\eg}{2^{n}}} \]
	זו בבירור קבוצה פתוחה, $\Q \subset A_\eg$, ו־$A_\eg \subsetneq \R$ (כי מידתה לכל היותר $2\eg$ סופית מתת־סיגמה אדטיביות. ראינו כבר את זה בעבר). זו קבוצה פתוחה, צפופה ו''קטנה``. נוכל להקטין אותה אפילו עוד: 
	\[ A = \bigcap_{k = 1}^{\infty} A_{\frac{1}{k}} \]
	זו קבוצה אפילו יותר קטנה (כי $m(A) \le m(A_{\frac{1}{k}}) \le \frac{2}{k} \to 0$), צפופה (כי $\Q\subset A$), והיא גם מעוצמת הרצף (ממשפט הקטגוריה של בייר). הקבוצה $A$ היא קבוצת $G_\dg$. 
	\defi{תהא $A \subset X$ מרחב טופולוגי. $A$ תקרא $G_\dg$ אם קיימות $(O_i)_{i = 1}^{\infty}$ פתוחות כך ש־$A = \bigcap_{i = 1}^{\infty} O_i$, ו־$F_\sg$ אם קיימות $(F_i)_{i = 1}^{\infty}$ סגורות כך ש־$A = \bigcup_{i = 1}^{\infty}F_i$. }
	
	\exe{חיתוך של $\sg$־אלגבראות הוא $\sg$־אלגברה. }
	
	\defi{ה־$\sg$ של בורל, היא ה־$\sg$־אלגברה המינימלית שמכילה קבוצות פתוחות ב־$\R^{d}$. }
	\rmark{בלעז, Borel. }
	\rmark{ה־$\sg$־אלגברה של בורל מוגדרת ביחידות מכיוון שניתן לקחת את החיתוך של כל ה־$\sg$־אלגבראות ולקבל $\sg$־אלגברה מינימלית. }
	מסתבר שה־$\sg$־אלגברה של בורל קטנה יותר מה־$\sg$־אלגברה של לבג (קבוצת הקבוצות המדידות). 
	\clm{ה־$\sg$־אלגברה של בורל, היא הקבוצה המורכבת מהקבוצות הבאות: 
	\begin{itemize}
		\item קבוצות פתוחות
		\item קבוצות סגורות
		\item קבוצות $F_\sg$
		\item קבוצות $G_\dg$
	\end{itemize}}

	נרצה לנסח קריטריון למדידות של קבוצות ב־$A \subset \R^{d}$ שלא מערב ריצוף ספציפי של $\R^{d}$. 
	
	\begin{Theorem}[תנאי המדידות של Lebesgue]
		תהא $A \subset \R^{d}$ קבוצה. אז $A$ מדידה אמ''מ לכל $\eg > 0$ קיימות $G$ פתוחה ו־$F$ סגורה כך ש־$F \subset A \subset G \land m(G \setminus F) < \eg$. 
	\end{Theorem}\begin{proof}
		\begin{itemize}
			\item[$\implies$] רוצים להוכיח ש־$A \cap \Og_n$ מדידה לכל $n$. נבחר $\Og \cod \Og_n$, ונסמן $A' = A \cap \Og$. רוצים להראות ש־$A'$ מדידה ב־$\Og$, כלומר שיש לה קירוב אלמנטרי טוב. יהי $\eg > 0$, ומתנאי המשפט נמצא $F \subset A \subset G$ מתאימות. נתבונן ב־$F' = F \cap |og, G' = G \cap \Og$. ברור ש־$F', G'$ מדידות ב־$\Og$ (כי $G' \setminus \ \partial \Og$ פתוחה ו־$F' \setminus \partial \Og $ סגורה, והשפה זניחה). 
			
			חיתוך של מדידות מדיד ב־$\R^{d}$ ולכן $G', F'$ מדידות ב־$\R^{d}$. לכן ל־$F', G'$ יש קירובים אלמנטריים $P_\eg, Q_\eg \subset \Og$ כך ש־$m^*(F' \trio P_\eg) ,m^*(G' \trio Q_\eg) < \eg$. נבחין ש־$P_\eg$ קירוב טוב: 
			\[ m^*(A' \trio P_\eg) \le m^*(\underbrace{A' \trio F'}_{A' \setminus F'}) + m^*(F' \trio P_\eg) \le m(A \setminus F) + \eg \le m(G \setminus F) + \eg \le 2\eg \]
			\item[$\impliedby$]נתון ש־$A$ מדידה, ולכל $\eg > 0$ נרצה למצוא $F, G$ המקיימים את התנאים. נגדיר (לצורך ההוכחה הזו) פנים של מלבן אלמנטרי־פתוח``. היה תרגיל לפיו מידה חיצונית אינווריאנטית ביחס ללקיחת פנים. 
			
			יהי $\eg > 0$. נבנה פתוחה $G$ כך ש־$m(G \setminus A) < \eg$. לכל $n \in \N$ ידוע ש־$m(A \cap \Og_n) = m^*(A \cap \Og_n)$ ולכן מהגדרת המידה החיצונית קיימים מלבנים אלמנטריים פתוחים $P_{n}^{(1)}, P_{n}^{(2)} \dots \subset \Og_n$ שקרובים לאינפימום כרצוננו ובפרט בקצב מעריכי (ישירות מהגדרת $\inf$ וממדידות), כלומר:
			\begin{align*}
				A \cap \Og = \bigcup_{i = 1}^{\infty}P_n^{(i)} && \sum_{i = 1}^{\infty}m(P_{n}^{(i)}) \le \frac{\eg}{2^{n}} + m(A \cap \Og_n)
			\end{align*}
			נשים לב ש־: 
			\[ \sum_{i = 1}^{\infty}m\cl{P_n^{(i)} \setminus A} \le \frac{\eg}{2^{n}} \]
			כי: 
			\[ m(P_{n}^{i} \setminus A) = m(P_n^{(i)}) - m(P_n^{(i)} \cap A) \]
			(הכל מתכנס בהחלט) דהיינו: 
			\[ \sum_{i = 1}^{\infty} \cl{m(P_n^{(i)}) - m(P_n^{(i)} \cap A)} = \frac{\eg}{2^{n}} + m(A \cap \Og_n) - \underbrace{\sum_{i = 1}^{\infty}m(P_n^{(i)} \cap A)}_{\mathclap{m\cl{\bigcup^{\infty}_{i = 1}(P_n^{(i)}\cap A)}}} = \frac{\eg}{2^{n}} + \cancel{m(A \cap \Og_n) - m(A \cap \Og_n)} \]
			(תנסו לצייר את זה, זה אמור להיות יחסית אינטואיטיבי). 
			עתה נגדיר: 
			\[ G \cod \bigcup_{n, i}P_n^{(i)} \]
			זו פתוחה (איחוד פתוחות), שמכילה את $A$ (כי לכל $n$ מתקיים $G \supset \bigcup_{i = 1} P_n^{(i)} \supset A \cap \Og_n$ ובאיחוד מוכלל על $n$ זה אכן מכיל את $A$). נחשב את $m(G \setminus A)$: 
			\[ m(G \setminus A) = m\!\cl{\bigcup_{n, i}P_n^{(i)}\setminus A} \le \sum_{n, i}m(P_n^{(i)}\setminus A) \le \sum_{n =1}^{\infty}\frac{\eg}{2^{n}} = \eg \]
			
			איך נמצא את $F$? נחזור על התהליך בעבור $\R^{d} \setminus A$, ונקב ל$G'$ פתוחה כך ש־$\R^{d} \setminus \subset G'$ וכן $m(G' \setminus (\R^{d} \setminus A))) < \eg$ ומכאן קל לראות ש־$F = \R^{d} \setminus G'$ מקיימת את הנדרש. 
			
		\end{itemize}
	\end{proof}

	\cola{תהא $A \subset \R^{d}$ מדידה. אז: 
	\begin{align*}
		m(A) &=\inf\ccb{m(G) \vert A \subset G \land G \ \text{\en{open}}} \\
		&= \inf\ccb{m(F) \vert F \subset A \land F \ \text{\en{closed}}}
		\end{align*}}\begin{proof}
		ברור ש־$m(A) \le \inf \ccb{\cdots}$. מהצד השני, לכל $\eg > 0$, $G$ פתוחה ו־$F$ סגורה, נקבל ש־: 
		\[ m(G \setminus A) \le m(G \setminus P) < \eg \]
		ולכן $m(G) = m(A) + m(G \setminus A) \le m(A) + \eg$ וסיימנו. הסופרמום באופן דומה. 
	\end{proof}
	
	
	\exe{לכל $A \subset \R^{d}$ מדידה, מתקיים ש־: 
	\[ m(A) = \sup\ccb{m(K) \mid K \subset A \land K \ \text{\en{compact}}} \]}
	
	\cola{לכל מדידה $A \subset \R^{d}$ קיימת $S \subset \R^{d}$ שמכילה את $A$ והיא $G_\dg$, כך ש־$m(S \setminus A)= 0$. }\begin{proof}
		לכל $n$ יש $G_n$ פתוחה שמכילה את $A$ ו־$m(G_n \setminus A) \le \frac{1}{n}$, ונגדיר: 
		\[ S = \bigcap_{i = 1}^{\infty}G_n \]
		מהגדרה היא $G_n$ ו־: 
		\[ m(S \setminus A) \le m(G_n \setminus A) \le \frac{1}{n} \toinf 0 \]
	\end{proof}
	
	בכך סיימנו את בניית מידת לבג ב־$\R^{d}$. 
	
	\defi{רציפות: עבור קבוצות $A_1, A_2 \dots $ כותבים ש־$A_n \nea A$ אם $A_i \subset A_{i + 1} \land \bigcup^{\infty}_{i = 1} A_i = A$. }
	
	נבחין במספר תכונות של מידת לבג שמצאנו: 
	\clm{נניח ש־$A \subset \R^{d}$ מדידה. אז: 
	\begin{itemize}
		\item אם $x \in \R^{d}$, אז $A + x$ מדידה. 
		\item אם $\lg \in \R$, אז $m(\lg A) =\sof{\lg}^{d} m(A)$ מדידה. 
		\item אם $T \co \R^{d} \to \R^{d}$ לינארית, אז: 
		\[ m(T(A)) = \sof{\det T} m(A) \]
		(הוכחה של זה בחדו''א 3)
		\item רציפות: אם $A_n \nea A$ וכולן מדידות, אז:
		\[ m(A) = \lim_{\mathclap{n \to \infty}} m(A_n) \]
		(ההוכחה מסיגמה אדטיביות תוך שימוש ב־$B_n = A_n \setminus A_{n - 1}$). יש תכונה דומה על קבוצה יורדת של סדרות. 
		\item שלמות: אם $A$ מדידה ו־$m^*(B \trio A) =0$ אז גם $B$ מדידה. 
		
		(זה לא נכון בסיגמה־אלגברה של בורל)
	\end{itemize}}
	
	
	\subsubsection{בניית קבוצה לא מדידה}
	\begin{Theorem}
		קיימת קבוצה לא־מדידה (לבג) אמ''מ אקסיומת הבחירה (הרציפה). 
	\end{Theorem}
	\rmark{הטענה לעיל בעלת משמעות רק בתוך ZF. }
	\textbf{תזכורת: }אקסיומת הבחירה אומרת שבהינתן $\{A_\ag\}_{\ag \in I}$ רשת קבוצות לא ריקות, קיימת $f \co I \to \bigcup_{\ag \in I}A_\ag$ כך ש־$\forall \ag \in I \co f(\ag) \in A_\ag$. 
	
	\defi{עבור $x, y \in [0, 1]$ נגדיר יחס שקילות $x \sim y \iff x - y \in \Q$. }
	\lem{זה יחס שקילות. }
	\defi{נתבונן באוסף מחלקות השקילות $\{E_\ag\}_{\ag \in I}$. \textbf{נבחר} $X_\ag \in E_\ag$ לכל $\ag \in I$ ($X_\ag \co I \to \bigcup E_\ag$ פונקציית בחירה), ונגדיר $A = \ccb{X_\ag \mid \ag \in I}$ מערכת הנציגים. }
	\clm{\label{clm:unmesaure}$A \subset [0, 1]$ קבוצה לא מדידה (לבג). }
	\begin{Lemma}[ראשונה]
		בהינתן $r, s \in \Q$ שונים מתקיים $(A + r) \cap (A + s) = \vn$. 
	\end{Lemma}\begin{proof}
		נניח בשלילה שישנו $x \in (A + r) \cap (A + s)$. אז (מהגדרה) ישנם $\ag, \ag' \in I$ כך ש־$x = x_\ag + r = x_{\ag'} + s$. לכן: 
		\[ x_\ag  - x_{\ag'} = s - r \in \Q \]
		מהגדרת מערכת הנציגים $x_{\ag} = x_{\ag'}$ ולכן $r = s$ בסתירה. 
	\end{proof}\begin{Lemma}[שנייה]
		\[ [0, 1] \subset \ \bigcup_{\mathclap{r \in \Q \cap [-1, 1]}} \ (A + r) \subset [-1, 2] \]
	\end{Lemma}\begin{proof}
		מצד אחד, $A \subset[0, 1]$ ולכן האיחוד מוכל ב־$[-1, 2]$. למה האיחוד מכיל את כל $[0, 1]$? כי לכל $x \in [0, 1]$, $x$ נמצאת באיזושהי מחלקת שקילות $E_\ag \in I$. ואז $x - x_\ag \in \Q$, ושתיהן ב־$[0, 1]$ ולכן ההפרש ביניהן הוא ב־$[-1, 1]$ (וכן רציונלי). בכך סיימנו. 
	\end{proof}
	\begin{proof}[הוכחת \ref{clm:unmesaure}]
		נניח בשלילה שהיא מדידה. אז $m(A) \le m([-1, 2]) = 3$. אך מ־$\sg$־אדטיביות (שמתקיימת למדידות) ידוע ש־: 
		\[ 1 = m([0, 1]) \le m\cl{\bigcup_{r \in \Q\cap[-1, 1]}\dequad A + r} \le m([-1, 2]) = 3 \]
		כך מלמה 1 האיחוד הוא זר (ובן מנייה), לכן: 
		\[ m{\cl{\bigcup_{r \in \Q\cap[-1, 1]}\dequad A + r} } = \sum_{\mathclap{r \in \Q\cap[-1, 1]}}(A + r) = \sum_{\mathclap{r \in \Q\cap[-1, 1]}}m(A) \]
		קיבלנו סכום בן מנייה של אותו המספר, לכן אם $m(A) = 0$ אז הטור מתכנס ל־$0$, אם $m(A) > 0$ הטור מתבדר. לכן הטור מתכנס ל־$0$ בסתירה לכך שהוא מתכנס לערך ב־$[1, 3]$. 
	\end{proof}
	
	מעניין לשאול מה $m^*(A)$. משום ש־$A$ לא מדידה אז $m(A) \neq 0$ (אחרת היא הייתדה מדידה), כלומר $0 < m^*(A) \le 1$. נסמן $0 < c = m^*(A)$. קיבלנו, בתוך $[-1, 2]$, אינסוף קבוצות זרות, שלכל האחת מידה חיצונית $0 < c$. 
	
	הבנייה הכי מפורסמת לקבוצות לא מדידות נקראת פרדוקס בנך־טרסקי. היא לא עובדת ב־$\R^{d}$ אבל היא עובדת ב־$\R^{n}$ לכל $n > 2$. אפשר גם לנסח אותו בלי להגדיר את מושג המידה: הרעיון הוא לחלק את כדור היחידה ב־$\R^{n}$ לקבוצות זרות, כך שעד לכדי איזומטריה ניתן לאחד את הקבוצות הזרות הללו לשני כדורים שונים. הדבר ה''נחמד`` הוא שלא נדרשים כאן להגדיר את מושג המידה.

	\section[מרחבי מידה אבסטרקטיים]{\en{Abstract Mesaure Theory}}
	\subsection{הגדרת מרחב מידה והשריית מידה}
	הרעיון: להבין באיזה תכונות בדיוק השתמשנו בהן ואז להכליל אותן. 
	
	\defi{תהא $\Og$ קבוצה. $\fc \subset \ps(\Og)$ הוא \textit{$\sg$־אלגברה} אם: 
	\begin{itemize}
		\item \hfil $\vn \in \fc$
		\item \hfil $A \in \fc \implies \Og \setminus A \in \fc$
		\item \hfil $\disty (A_i)_{i = 1}^{\infty} \in \fc \implies \bigcup_{i = 1}^{\infty}A_i \in \fc$
	\end{itemize}}
	\defi{ל־$(\Og, \fc, \mu)$ נקראת \textit{מרחב מידה} אם: 
	\begin{enumerate}
		\item $\fc$ היא $\sg$־אלגברה של ת''קים של $\Og$. קבוצות ב־$\fc$ יקראו \textit{קבוצות מדידות}. 
		\item $\mu \co \fc \to [0, +\infty]$ והיא $\sg$־אדטיבית, כלומר לכל $A_1, A_2 \dots \in \fc$ זרות, מתקיים: 
		\[ \mu\!\cl{\bigcup_{i =1}^{\infty}A_n} = \sum_{n = 1}^{\infty}\mu(A_n) \]
	\end{enumerate}}\ 
	\cola{אם $\mu \neq \infty$ אז $\mu(\vn) = 0$. זה נובע ישירות מתכונה $2$. }
	\defi{אם: 
	\begin{itemize}
		\item $\mu(\Og) = 1$, המרחב נקרא \textit{מרחב הסתברות}. 
		\item אם $\mu(\Og) < \infty$, המרחב נקרא \textit{מרחב סופי}. 
		\item אם לכל $A \in \fc$, מתקיים ש־$\mu(A) = 0 \implies \forall B \subset A \co B \in \fc$, אז המרחב נקרא \textit{שלם}. 
	\end{itemize}}
	
	\nthlec{3}{4}
	
	\textbf{דוגמאות. }
	\begin{itemize}
		\item $\Og \cod \R^{d}$ עם קבוצת הקבוצות המדידות ומידת לבג $m$, הוא מרחב מידה שלם. זו ההגדרה הכללית ביותר המקובלת על מתמטיקאים למושג המידה ב־$\R^{d}$. 
		\item עבור $\Og = \Z$ (או כל קבוצת בת־מנייה). בהינתן משקלים $\{w_n\}_{n \in \Z}$ אי־שליליים, נגדיר $\fc = 2^{\Z}$ עם המידה $\mu(A) = \sum_{n \in A}w_n$. 
		
		ברור שזו סיגמא אלגברה (פשוט כל המספרים כולם), ובגלל שהמשקולות אי־שליליים אין שאלה לגבי סדר הסכימה אז ברור ש־$\mu$ היא $\sg$־אדטיבית. 
		
		למידה כזו קוראים \textit{מידה דיסקרטית}, ולעיתים מסמנים $\mu(A) =\sum_{n \in \Z} w_n\dg_n$ כאשר $\dg_n$ נקראת \textit{מידה אטומית}, ומוגדרת להיות: 
		\[ \dg_n(A) = \begin{cases}
			1 & n \in A \\
			0 & \other
		\end{cases} \]
		\item דוגמה נוספת, שנוודא שהיא מגודרת היטב בהמשך: נניח שיש פונקציה $\rho \co \R^{3} \to [0, \infty)$ שמתארת ''צפיפות מסה``. היא משרה מידה על קבוצות בורל (ואפילו לבג) שברמה האינטואיטיבית הוא: 
		\[ \mu_\rho(A) = \int_A \rho \]
		את אינטגרל לבג נגדיר מאוחר יותר. 
	\end{itemize}
	
	איך עובדים עם מידה כללית? 
	\begin{Claim}
		\begin{enumerate}
			\item מתקיים $\mu\cl{\bigcup_{n = 1}^{\infty}A_n} \le \sum_{n = 1}^{\infty}\mu(A_n)$.
			\item נניח ש־$A_n \nea A$ מיצוי, אז $\mu(A_n) \toinf \mu(A)$
		\end{enumerate}
	\end{Claim}
	\begin{proof}
		\begin{enumerate}
			\item 	אותה ההוכחה מלפני שבועיים – הרי לא השתמשנו בה בתכונות נוספות. כלומר נוכל לבוא ולהגדיר: 
			\[ A_0' = A_n \setminus \cl{\bigcup_{n = 1}^{\infty}A_n} \]
			כדי לקבל קבוצות זרות, ואז: 
			\[ \mu\cl{\bigcup_{n = 1}^{\infty}A_n} = \mu\!\cl{\bigcup_{n = 1}^{\infty}A_n'} = \sum_{n = 1}^{\infty}\underbrace{\mu(A_n')}_{\le \mu(A_n)} \]
			\item נגדיר $B_1 = A_1, \ B_n = A_n \setminus A_{n + 1} \ (n \ge 2)$, אז: 
			\[ \mu(A_n) = \mu\!\cl{\bigcup_{k = 1}^{n}B_k} = \sum_{k = 1}^{n}\mu(B_k) \toinf \sum_{k = 1}^{\infty}\mu(B_k) = \mu\!\cl{\bigcup_{k = 1}^{\infty}B_k} = \mu(A) \]
		\end{enumerate}
	\end{proof}
	ה־$\sg$־אדטיביות היא התכונה החשובה של המידה, ובאמצעותה לרוב ניגש להוכיח את רוב התכונות האקזויוטיות של המידה. נזכר בקצרה אין בנינו את מידת לבג: 
	\begin{itemize}
		\item התחלנו מקבוצות אלמנטריות, עליהן יש מידה אדטיבית־סופית, שהייתה גם תת־סיגמה אדטיבית
		\item הגדרנו מידה חיצונית
		\item הגדרנו קבוצות מדידות, וכו'
	\end{itemize}
	
	אפשר להכליל את התהליך הזה. 
	
	\begin{Theorem}[גרסה של משפט Carathodory]\label{theo:carathodory}
		תהי $\Og$ קבוצה, ותהי $\fc \in 2^{\Og}$ אלגברה של קבוצות (''אלמנטריות``). נניח ש־: 
		\begin{itemize}
			\item $\mu$ סופית, כלומר $\mu(\Og) < \infty$
			\item $\mu$ תת־סיגמה אדטיבית, כלומר אם $A_1, A_2 \dots \in \fc$ ואם $A \subset \bigcup A_i$, אז $\mu(A) \le \sum \mu(A_n)$. 
		\end{itemize}
		אז קיימת \ס־אלגברה שמכילה את $\fc$, והרחבה של $\mu$ למידה \ס־אדטיבית שלמה המוגדרת על ה־\ס־אלגברה הזו. 
	\end{Theorem}\begin{proof}
		ספוילר: חוזרים על השבועיים הראשונים של הקורס. 
		
		ראשית מגדירים מידה חיצונית: לכל $A \subset \Og$, נסמן: 
		\[ \mu^*(A) \cod \inf \ccb{\sum_{i = 1}^{\infty}\mu(\fc_i) \ \middle\vert \ \forall i\co \fc_i \in \fc \land A \subset \bigcup_{i = 1}^{\infty}\fc_i} \]
		אותה ההוכחה בעבור מידת לבג ב־$\R^{d}$ תעבוד כאן, כדי להראות ש־$\mu^*$ תת־\ס־אדטיבית, וש־$\forall A \in \fc \co \mu^*(A) = \mu(A)$. 
		
		לאחר מכן נגדיר קבוצה מדידה באופן הבא: $A \subset \Og$ תוגדר להיות מדידה אמ''מ לכל $\eg > 0$ יש $E_\eg \in \fc$ כך ש־$m^*(A \trio E_\eg) < \eg$. מראים שקבוצת הקבוצות המדידות היא \ס־אלגברה, ושצמצום $\mu^*$ ל־\ס־אלגברה הזו היא מידה $\sg$־אדטיבית ושלמה. 
	\end{proof}

	כזכור, ה־$\sg$־אלגברה של \textit{בורל} היא המינימלית המכילה קטעים פתוחים וקבוצות פתוחות (במינימלית, הכוונה היא שחיתוך כל ה\ס אלגבראות שמכילות את הקטעים הפתוחים נותן לנו את ה\ס אלגברה של בורל. זה מוגדר היטב כי חיתוך \ס־אלגבראות הוא \ס־אלגברה). ממשפט קרתאודורי, אם מתחילים ממידה אדטיבית־סופית ותת־אדטיבית על קבוצות אלמנטריות ב־$\R$ או $\R^{2}$ (כלשהי! לאו דווקא המידה הרגילה שהגדרנו על איחוד מלבנים), אזי מובטחת לנו הרחבה לקבוצות בורל. לא מובטחת הרחבה לקבוצות לבג. 
	
	\begin{Theorem}[יחידות]
		הרחבת קאתאודורי הנ''ל ל־\ס־אלגברה המינימלית שמכילה את $\fc$, יחידה. 
	\end{Theorem}\begin{proof}
		נסמן ב־$\sg(\fc)$ את ה־$\sg$־אלגברה המינימלית שמכילה את $\fc$. נניח ש־$\nu$ מידה \ס־אדטיבית על $\sg(G)$, כך ש־$\forall A \in \fc \co \nu(A) = \mu(A)$. נרצה להראות שגם $\forall A \in \sg(F) \co \nu(A) = \mu^*(A)$. 
		
		תהא $A \in \sg(\fc)$. 
		\begin{itemize}
			\item ראשית נראה $\nu(A) \le \mu^*(A)$. נדרוש להראות ש־$\nu(A) \le \sum_{i = 1}^{\infty}\mu(F_i)$ לכל $F_1, F_2 \dots \in \fc$ שאיחודן מכסה את $A$ (מהגדרה). ידוע $\forall i \co \nu(F_i) = \mu(F_i)$ ו־$\sg$־אדטיביות סופית נקבל ש־: 
			\[ \nu(A) \le \sum_{i = 1}^{\infty}\nu(F_i) = \sum_{i = 1}^{\infty}\mu(\fc_i) \]
			\item לכיוון השני, נוכל להראות ש־$\nu(A) \ge \mu^*(A)$ בקלות באמצעות לקיחת המשלים. כזכור, $\mu(\Og) < \infty$. לכל $A \in \sg(F)$ מתקיים ש־: 
			\[ \begin{WithArrows}
				\mu^*(A) &= \mu^*(A) + \mu^*(\Og \setminus A) \Arrow{מהכיוון הקודם}\\
				&\ge \nu(A) + \nu(\Og \setminus A) \Arrow{אדטיביות סופית}\\
				&\ge \nu(\Og) = \mu(\Og)
			\end{WithArrows} \]
			לכן הכל שוויונות ונסיק $\mu^*(A) = \nu(A)$. 
		\end{itemize}\envendproof
	\end{proof}
	
	עתה כשיש יחידות, נוכל להגדיר את ההרחבה לעיל באופן מוגדר היטב. 
	\defi{להרחבה הנ''ל של $\mu$ קוראים \textit{הרחבת קראתאודורי של $\mu$}. }
	\cola{בהינתן $\mu_1, \mu_2$ מידות \ס־אדטיביות של $\sg(F)$, כדי להראות $\mu_1 = \mu_2$ מספיק להראות ש־$\mu_1(A) = \mu_2(A)$ לכל $A \in \fc$. }
	המסקנה הנ''ל נכונה, כי אם המידות מזדהות על גבי האלגברה $\fc$ אז הן מזדהות גם על כל הרחבה שלה. 
	
	\textbf{דוגמה. }מידת לבג על מעגל: 
	\begin{itemize}
		\item אפשר להגדיר את הקבוצות האלמנטריות להיות ''קשתות פתוחות משמאל וסגורות מימין`` 
		\item האלמנטריות אלגברה, מגדירים עליה מידה אטיבית בפשטות
		\item כדי להשתמש בהרחבת קרטאודורי, צריך להוכיח תת־\ס־אדטיביות (במקרה הזה זה נובע מהיינה בורל). 
		\item הקטע המגניב: המשפט לא דורש שהאלגברה תהיה \ס־אלגברה. הוא מרחיב את האלגברה בעצמו. 
	\end{itemize}
	
	היה לנו תהליך נוסף לבניית מידת לבג. כדי להרחיב את קבוצת הקבוצות המדידות שלנו לכל המרחב, ריצפנו את המישור בעותקים של ה־$\Og$־ות. 
	\theo{\label{theo:tiling}נניח ש־$\Og = \biguplus_{n = 1}^{\infty} \Og_n$. נניח ש־$(\Og_n, \fc_n, \mu_n)$ מרחב מידה סופי לכל $n \in \N$. נגדיר \ס־אלגברה $\fc \subset 2^{\Og}$ ו־$\mu$ מידה על $\fc$ לפי: 
	\begin{itemize}
		\item עבור $A \subset \Og$, נאמר ש־$A \in \fc \iff \forall n \in \N \co A \cap \Og_n \in \fc_n$
		\item אם $A \in \fc$, אז נגדיר $\mu(A) = \sum_{n = 1}^{\infty}\mu_n(A \cap \Og_n)$
	\end{itemize}
	אזי $(\Og, \fc, \mu)$ הוא מרחב מידה. }
	
	חשוב שהכל בן מנייה – אם היינו מנסים לעשות את התהליך על זה באמצעות איחוד מעוצמת האפס, נקבל דברים מעוותים (לדוגמה: $\R$ כסכום של סינגלטונים). 
	
	\begin{proof}
		ע''ע המשפט בפרק הקודם
	\end{proof}
	
	\defi{מרחב מידה $(\Og, \fc, \mu)$ נקרא \textit{\ס־סופי} אם קיים פירוק $\Og = \biguplus \Og_n$, כך ש־$\Og_n \in \fc$ go $\mu(\Og_n) < \infty$ (מדידות עם מידה סופית). }
	
	\subsection{דחיפה קדימה של מידות}
	\rmark{באנגלית, push forward}
	\defi{נניח $(\Og, \fc, \mu)$ מרחב מידה סופי, ו־$\tl \Og$ קבוצה, כך ש־$T \co \Og \to \tl \Og $ העתקה. מגדירים \ס־אלגברה $\tl F \subset 2^{\tl \Og}$, ו־$\tl \mu$ לפי: 
	\begin{itemize}
		\item \hfil $\tl F = \ccb{A \in \tl \Og \co T\op(A) \in \fc}$
		\item \hfil $\forall A \in \tl F \co \tl \mu(A) = \mu(T\op (A))$
	\end{itemize}
	המרחב $(\tl \Og, \tl \fc, \tl \mu)$ נקרא \textit{הדחיפה קדימה של $\mu$}. }
	
	\rmark{\label{rmark:reverseTrans}אנחנו משתמשים כאן בכבדות שההעתקה הפוכה (הפעולה $T\op$ לכל פונקציה, לאו דווקא חח''ע) משמרת את כל התכונות שלנו של תקב''צ – משלים, איחוד, חיתוך וכו' (בניגוד להעתקה ''רגילה``). אנחנו משתמשים בזה בכבדות. }
	
	\theo{הדחיפה קדימה הוא מרחב מידה. }
	
	המשפט נובע ישירות מהערה \ref{rmark:reverseTrans}. 
	
	\textbf{דוגמאות. }
	\begin{enumerate}
		\item עבור מרחב המידה $I^{2}$ עם מידת לבג, ו־$T(x, y) = x$ היטל, $T$ דוחפת קדימה ומשרה את מידת לבג על $I$ לתוך $T(I^{2}) = I$. זאת כי $T\op(A) = A \times [0, 1]$ מקיים $\tl \mu(A) = m_2(A \times [0, 1]) = m_1(A)$. לכן $m_2 = \tl \mu$ (הן מזדהות על אלמנטריות, מספיק לוודא זאת כדי להראות שוויון מידות). 
		\item עבור $\Og = [0, 1] \setminus \Q_{\text{\en{Dyadic}}}$ עם מידת לבג. (מספר דייאדי זה מספר בעל פיתוח סופי בבסיס $2$. יש להם שני ייצוגים כאלו. לשאר המספרים יש ייצגו אין־סופי ויחיד. ) נגדיר: 
		\begin{align*}
			T \co \Og \to \{0, 1\}^{\N} && T \!\cl{\sum_{i = 1}^{\infty}\frac{a_i}{2^{i}}} = (a_1, a_2 \dots)
		\end{align*}
		ההעתקה מוגדרת היטב כי הייצוג הבינארי יחיד למספרים שאינם דיאדיים. נבחין שיש רק מספר בן־מנייה של מספרים דיאדיים עם פיתוח עשרוני סופי, ולכן ''הוצאנו`` מ־$[0, 1]$ קבוצה זניחה כדי לקבל את $\Og$. לכן $m(\Og)  = 1$. 
		
		איך נראית הדחיפה קדימה? מקבלים מידת הסתברות ואיזושהי \ס־אלגברה על אוסף הסדרות. זו מידת הסתברות של הטלות מטבע הוגן (לדוגמה, אוסף כל הסדרות שמתחילות ב־$0,1, 1,1$ הוא ממידה $\frac{1}{16}$). 
	\end{enumerate}
	
	\subsection{מידות לבג־סטילטייס}
	\rmark{בלעז, Lebesgue-Stielties}
	
	מדובר במשפחה של מידות על $\R$ שמושרות ע''י פונקציה $f \co \R \to \R$ שהיא לא־יורדת (בשום מקום, כלומר היא מונוטונית עולה חלש) ורציפה משמאל (ייתכנו נקודות אי־רציפות מסדר ראשון). 
	מטרתנו תהיה לבנות מידה באמצעות $f$. נשתמש בגישת קרתאודורי. נתחיל מקבוצות פשוטות: עבור $a < b$, נגדיר: 
	\[ \mu_f([a, b)) = f(b) - f(a) \]
	
	כמובן נתחיל ב־$\Og = [0, 1]$ ואז נרצף (בהתאם למשפט שכבר ניסחנו). קבוצה אלמנטרית תהיה איחוד של מלבנים פשוטים, וזו אלגברה של קבוצות ב־$I$. באופן לא מפתיע, עבור $A$ איחוד זר של $[a_i, b_i)$ נגדיר: 
	\[ \mu(A) \cod \sum_{i = 1}^{n}\mu([f(b_i) - f(a_i)]) \]
	צ.ל. שההגדרה לא תלויה בפירוק לאיחוד זר של קטעים אלמנטריים (כדי להראות שזה מוגדר היטב). זה פשוט מאוד מטור טלסקופי, כלומר אינדוקציה על זה ש־: 
	\[ \mu_f([a, b)) =f(b) -f(a) = (f(b) - f(c)) + (f(c) - f(a)) = \mu_f([a, c)) + \mu_f([c, b)) \]
	נוסף על כך צריך להראות ש־$\mu_f$ אדטיבית סופית. זה ישירות מההגדרה. 
	
	\clm{$\mu_f$ המוגדרת לעיל תת־אדטיבית. }
	שימו לב שטרם השתמשנו ברציפות משמאל. 
	\clm{\label{clm:prefunctionmeasure}אם $f$ לא יורדת ורציפה משמאל, אז $\mu_f$ על הקבוצה האלמנטריות היא תת־\ס־אדטיבית. }
	\lem{יהי $[a, b) \subset I$. אזי לכל $\eg > 0$ קיימת $\dg > 0$ כך ש־: 
	\[ \mu_f\cl{[a, b - \dg)} \ge \mu([a, b]) - \eg \quad\land\quad \mu_f([a - \dg, b)) \le \mu_f([a, b)) + \eg \]\ }\begin{proof}
		מרציפות משמאל:
		\[ \lim_{\dg \to 0^{+}}\mu_f([a, b - \dg)) = \lim_{\dg \to 0^{+}}f(b - d) - f(a) = f(b) - f(a) = \mu_f([a, b)) \]
		מהגדרת הגבול, קיימת $\dg$ כנדרש. ובאופן דומה מהצד השני $\lim_{\dg \to 0^+} \mu_f([a - \dg, b)) = \mu([a, b))$, נבחר את המינימום. 
	\end{proof}
	\lem{לכל אלמנטריות $A \subset [0, 1)$, ולכל $\eg > 0$, קיימות $A', \tl A \subset [0, 1)$ כך ש־: 
	\[ \Cl A' \subset A \subset \Int  \tl A \quad \land \quad \mu_f(A' \setminus \tl A) < \eg \]\ }\begin{proof}
		ישירות מהלמה הקודמת: ''קצת מקטינים מימיןאת הקטעים ליצירת $A'$, וקצת מגדילים משמאל ליצירת $\tl A$``. 
	\end{proof}
	\begin{proof}[הוכחת \ref{clm:prefunctionmeasure}]
		נתון $A \subset \bigcup_{n = 1}^{\infty}A_n$. יהי $\eg > 0$. מספיק להראות ש־$\mu_f(A) \le \sum_{n = 1}^{\infty} \mu_f(A) + \eg$. 
		
		צריך כיסוי פתוח של קומפקט, אז נשתמש בינה בורל. מלמה $1$ קיימת $A'$ אלמנטרית כך ש‏־$\Cl A' \subset A \land \mu_f(A' \setminus A) = \frac{\eg}{2}$. לכל $n$, קיימת $\tl A_n$ כך ש־$A_n \subset \Int(\tl A_n)$ וכן $\mu_f(\tl A_n \setminus A_n) < \frac{\eg}{2^{n + 1}}$. עתה: 
		\[ \Cl A' \subset A \subset \bigcup_{n = 1}^{\infty} \Int (\tl A_n) \]
		נבחין ש־$\tl A_n$ פתוחות ו־$\Cl A'$ קומפקטי. מהיינה בורל יש $N$ כך ש־$\Cl A' \subset \bigcup_{r = 1}^{N} \Int \tl A_r$, ולכן מאדטיביות סופית: 
		\[ \mu_f(A) \le \frac{\eg}{2} + \mu_f(A') \le \frac{\eg}{2}\mu\cl{\bigcup_{r = 1}^{N}\tl A_r} \le \frac{\eg}{2}\sum_{r = 1}^{N}\mu_f(\tl A_r) = \frac{\eg}{2} + \sum_{r = 1}^{N}\mu_f(A_r) \]
	\end{proof}
	\cola{לכן, מקרטאודורי (\ref{theo:carathodory}) וממשפט ההרחבה לכיסוי (\ref{theo:tiling}) קיבלנו מידה \ס־אדטיבית, המכילה את כל קבוצות בורל (כי היא מכילה את הקטעים החצי־פתוחים, ובורל המינימלית שמכילה אותם). }
	\defi{המידה לעיל נקראת מידת לבג־סטילטייס. }
	
	
	\textbf{דוגמאות. }
	\begin{enumerate}
		\item עבור $f(x) = x$, אז $\mu_{f}([a, b)) = b - a$, ולכן $\mu_f$ מזדהה עם מידת לבג. 
		\item עבור $f(x) = \begin{cases}
			0 & x \le 0 \\
			1 & x > 0
		\end{cases}$
		פונקציית המדרגה. נבחין ש־: 
		\[ \mu_f([a, b)) = f(b) - f(a) = \begin{cases}
			1 & 0 \in [a, b) \\
			0 & \other
		\end{cases} \]
		אז לכל אלמנטרית $A \subset I$, מתקיים: 
		\[ \mu_f(A) = \begin{cases}
			1 & 0 \in A \\
			0 & \other
		\end{cases} \]
		למעשה זה גם נכון לכל $A$ בורל, שכן זו מידה המזדהה עם האלמנטריות. למידה כזו קוראים \textit{מידה אטומית}. 
		
		\defi{למידה המושרה ע''י פונקציית המדרגה ב־$r$ קוראים \textit{מידה אטומית} ומסמנים אותה ב־$\dg_r$. מתקיים ש־: 
			\[ \dg_r(\{r\}) = 1 \quad\land\quad \dg_r(\R \setminus \{r\}) = 0 \]\eqs }
			
		בסיגמה אלגברה שמושרית ע''י המידה האטומית, כל קבוצה היא מדידה. 
		
		
		
		\item נניח ש־$f \in C^{1}$. נסמן $\rho = f'$. אז: 
		\[ \mu_f([a, b)) = f(b) - f(a) = \int^{b}_a\dequad \ f'(x)\dx = \int^{b}_a \dequad\  \rho(x)\dx \]
		(אלו, בינתיים, אינטגרלי רימן). ל־$\rho$ קוראים \textit{פונקציית צפיפות}. כאשר נלמד על אינטגרל לבג, נבחין ש־$\mu_f(A) = \int_A \rho$. לדוגמה, אם $\rho(t) = \frac{1}{\sqrt{2\pi}} e^{-t/2}$, אז למידה $\rho_f$ שמקבלים קוראים \textit{המידה הנורמלית}/\textit{סטנדרטית}. 
		\item \ \!\! [הרחבה בתרגול] 
		\defi{\textit{פונקציית קנטור} היא הפונקציה הבאה: 
		\[ f\cl{\sum \frac{a_i}{3_i}} = \sum \frac{a_i}{2_i} \]\eqsw}
		$f$ היא רציפה, $\mu_f(\cc) = 1 \land \mu_f(\R \setminus \cc) = 0$, ול־$f$ אין שום אטום ,כלומר $\mu_f\cl{\{a\}} = 0$. כלומר זוהי קבוצה שנתמכת על קנטור בלי אטומים. 
		
		זו מידה סינגולרית למידת לבג, כלומר היא נתמכת על קבוצה שביחס ללבג היא ממידה אפס. 
	\end{enumerate}
	
	\rmark{כל מידת בורל (כלומר מידה שקבוצות בורל מדידות בה) היא מידת בורל־סטיליטייס של איזושהי $f \co \R \to \R$. }\begin{proof}
		בהינתן מידת בורל $\mu_f$, נגדיר: 
		\[ f(t) \cod \mu([-\infty, t)) \]
		אכן: 
		\begin{itemize}
			\item $f$ לא יורדת, שכן $s < t$ גורר $[-\inft, s) \subset[-\inft, t)$ ויש תת־אדטיביות למידה. 
			\item $f$ רציפה משמאל כי: 
			\[ \lim_{n \to \inft, \eg_n \to 0}f(t - \eg_n) = \lim \mu((-\infty, t - \eg_n)) = \mu\cl{(-\infty, t)} = f(t) \]
%			TODO
		\end{itemize}
		ו־$\mu_f$ מתלכדת עם $\mu$ על אלמנטריות, ולכן שוות. 
	\end{proof}

	\section[פונקציות ממשיות מדידות]{\en{Real-Valued Lebesgue-Measurable Functions}}
	\subsection{פונקציות מדידות}
	
	נעבוד במרחב מידה $(\Og, \fc, \mu)$ ונתעניין בעיקר במידת לבג. 
	\defi{תהא $f \co \Og \to \R$. נגדיר $\{f < t\}$ להיות: 
	\[ \{f < t\} \cod \{x \in \Og \co f(x) < t\} = f\op\cl{(-\infty, t)} \]\eqs}
	\defi{פונקציה $f\co \Og \to \R$ נקראת \textit{מדידה} אם לכל $t \in \R$ הקבוצה $\{f < t\}$ מדידה (כלומר, שייכת ל־$\fc$). }
	
	\textbf{דוגמאות. }
	\begin{itemize}
		\item אם $A \subset \Og$ מדידה, הפונקציה האופיינית (האינדיקטור $\q_A(x)$) היא פונקציה מדידה. זאת כי: 
		\[ \{1_A < t\} = \begin{cases}
			\Og & t > 1 \\
			A^{c} & 0 < t \le 1 \\
			\vn & t \le 0
		\end{cases} \]
		כולן מדידות. 
		\item ב־$\R$, פונקציית דיריכלה שאינ אינטגרילית רימן, היא מדידה לבג כי $D = \q_\Q$ ו־$\Q$ מדידה לבג (הראנו שהאינדיקטורים מדידים)
		\item נניח ש־$f \co \R^{d} \to \R$ רציפה. אז היא מדידה. זאת כי רציפות שקולה לכך שתמונה הפוכה של פתוחה היא פתוחה. בפרט, מרציפות $f$, נובע ש־$\{f < t\} = f\op((-\infty, t))$ פתוחה ובפרט מדידה לבג. 
	\end{itemize}
	\clm{תהא $f \co \Og \to \R$ מדידה. אז לכל $t \in \R$, הקבוצות $\{f \le t\}, \ \{f > t\}, \ \{f = t\}, \ \{\sof f \le t\}$ מדידות. }
	\begin{proof}
		יודעים ש־$\{f < t\}$ מדידה לכל $t$, וש־$\fc$ \ס־אלגברה. רק נותר להציג את הקבוצות לעיל כאיחודים (לכל היותר בני מנייה) ומשלימים של קבוצות מהצורה $\{f < t\}$. 
		\begin{gather*}
			\{f \ge t\} = \Og \setminus \{f < t\} \\
			\{f \le t\} = \bigcap_{n = 1}^{\infty}\{f < t + \frac{1}{n}\} \\
			\{f = t\} = \{f \le t\} \cap \{f \ge t\} \\
			\{\sof f \le t\} = \{f \le t\} \cap \{f \ge -t\}
		\end{gather*}
		כל הקבוצות באגף שמאל מדידות. 
	\end{proof}
	ברור גם שזה עובד מהכיוון השני, כלומר אם היינו מגדירים את $f$ להיות מדידה אמ''מ, לדוגמה, $\{f \le t\}$, ההגדרה הייתה שקולה. 
	
	\nthlec{4}{5}. 
	\rmark{יש פונקציה בשם פונקציית Voltera, שהיא פונקציה $f \co \R\to \R$, גזירה, ליפשיצית, אבל $f'$ לא אינטגרבילית רימן. זה עושה בעיות במשפט היסודי של החדו''א. אבל אצלנו, באינטגרלי לבג, $f'$ כן תהיה מדידה. }
	\exe{נניח ש־$f \co \Og \to \R$ מדידה, אז לכל קבוצת בורל $A \subset \R$ הקבוצה $f\op(A)$ מדידה. (רמז: ה־push forward הוא \ס־אלגברה). }
	\rmark{אנקדוטה מהשיעור הקודם: לא ניתן לבנות קבוצה שאינה בורל אך מדידה לבג ללא שימוש באקסיומת הבחירה או גרסאות חלשות שלה. }
	\theo{נניח $f, g \co \Og \to \R$ מדידות. אזי $f + g$ מדידה. }\begin{proof}
		יהי $t \in \R$. צ''ל $\{f + g < t\}$ מדידה. נבחין ש־: 
		\[ \{f  +g < t\} = \bigcup_{\mathclap{\substack{\ag, \bg \in \Q \\ \ag + \bg < t}}} \big[{\{f < \ag \}\cap \{g < \bg\}}\big] \]
		ברור שמתקיים $\supset$ שכן אם $x \in \{f < \ag\} \cap \{y < \bg\}$ אז $f(x) < \ag \land y(x) < \bg$, כלומר $f(x) + g(x) < \ag + \bg < t$ וסה''כ $x \in\{f + g < t\}$. 
		
		מהצד השני ($\subset$), אם $x \in \{f + g < t\}$, אז מצפיפות הרציונליים ישנם $\ag, \bg \in \Q$ כך ש־: 
		\begin{gather*}
			f(x) < \ag < f(x) + \frac{t - f(x) - g(x)}{2} \\
			g(x) < \bg < g(x) + \frac{t - f(x) - g(x)}{2}
		\end{gather*}
		ולכן $\ag + \bg < t$, ו־$x \in \{f < \ag\} \cap \{y < \bg\}$ כנדרש. 
		
		סה''כ הצגנו את $\{f + g < t\}$ כאיחוד בן מנייה של קבוצות מדידות, ולכן היא מדידה גם היא (קבוצת המדידות היא \ס־אלגברה). 
	\end{proof}
	\cola{יהיו $f, g \co \Og \to \R$ מדידות. אז: 
	\begin{enumerate}[(A)]
		\item $f - g$ מדידה
		\item לכל $\ag \in \R$, גם $\ag f$ מדידה. 
		\item $f \cdot g$ מדידה. 
	\end{enumerate}}
	\begin{proof}
		\begin{enumerate}
			\item ברור ש־$-g$ מדידה שכן $\{g < t\} = \{g > -t\}$ ולכן גם $f - g$ מדיד. 
			\item לכל $c \in \R$: 
			\[ \{cg < t\} = \begin{cases}
				\{f < \frac{c}{t}\} & c \ge 0 \\
				\{f > \frac{c}{t}\} & c < 0
			\end{cases} \]
			הקבוצות מימין מדידות ולכן גם $fc$ מדידה. 
		\end{enumerate}
		\item ראשית נראה ש־$f^{2}$ מדידה: 
		\[ \{f^{2} < t\} = \begin{cases}
			\vn & t \le 0 \\
			\{f < \sqrt t\} \cap \{f > -\sqrt t\} & t > 0
		\end{cases} \]
		שתיהן מדידות. לכן $f^{2}$ מדידה. נעשה פולריזציה ממשית: 
		\[ fg = \frac{(f + g)^{2} - (f - g)^{2}}{4} \]
		ולכן מדידה. 
	\end{proof}
	
	\exe{אם $f\co \Og \to \R$ מדידה ו־$\phi \co \R\to \R$ רציפה (מספיק מדידה בורל), אז $\phi \circ f$ מדידה. }
	
	\subsection{סדרות פונקציות מדידות}\ 
	
	\theo{יהיו $f_1, f_2 \dots \co \Og \to \R$ פונקציות מדידות. אזי: 
	\begin{enumerate}[(A)]
		\item $\Og \ni x \mapsto \sup_{n\in \N} f_n(x)$ מדידה. 
		\item $\Og \ni x \mapsto \inf_{n\in \N} f_n(x)$ מדידה. 
		\item $\Og \ni x \mapsto \limsup_{n\in \N} f_n(x)$ מדידה. 
		\item $\Og \ni x \mapsto \liminf_{n\in \N} f_n(x)$ מדידה. 
	\end{enumerate}}(
	\rmark{אין בעיה להתיר את הסדרות להיות מוגדרות במובן הרחב, כלומר מותר לפונקציה על $\Og$ לקבל את $\pm \inf$ (כדי שהסופרמום יהיה מוגדר היטב), ונאמר ש־$f \co \Og \to \R \cup\{\pm\infty\}$ מדידה אם $A =\{f = +\infty\}$ מדידה, $B = \{f = -\infty\}$ מדידה, ו־$f|_{\Og \setminus (A \cup B)}$ מדידה. באופן שקול, $f \co \Og \to [-\infty, +\infty]$ אם לכל $t \in \R$ הקבוצה $\{f < t\}$ מדידה. }
	\begin{proof}
		לכל $t \in \R$, מתקיים ש־: 
		\[ \{x \in \Og \mid \sup_n f_n(x) \le t\} = \bigcap_{i = 1}^{\infty}\{f \le t\} \]
		ולכן הסופרמום מדיד. האינפימום מדיד שכן: 
		\[ \inf_nf_n(x) = -\sup_n(-f_n(x)) \]
		נוכל להיעזר בסופרמום גם ל־$\limsup$:
		\[ x\mapsto \limsup_{n \to \infty} f_n(x) = \lim_{N\to\infty}\csb{\sup_{n \ge N}f_n(x)} = \inf_{N \in \N}\underbrace{\csb{\sup_{n \ge N}f_n(x)}}_{\ \! \text{$_ng$ מדידה}} \]
		בעבור $g_n(x) = \sup f_n(x)$. אינפימום של מדידות מדיד. באופן דומה על ה־$\liminf$. 
	\end{proof}
	
	\cola{נניח ש־$f_1, f_2 \co \Og \to \R \cup{\pm\infty}$ פונקציות מדידות. נניח שהגבול $\lim_{n \to \infty}f_n(x)$ במובן הרחב לכל $x$. אז $f(x) = \lim_{n \to \infty}f_n(x)$ מדידה. }
	\prooftrivial
	
	\textbf{דוגמה. }תהא $f \co \R \to \R$ פונקציה גזירה. נסמן ב־$f_n(x) = \frac{f\cl{x + \frac{1}{n}} - f(x)}{1/n}$. אז לכל $x \in \R$, $f_n(x) \toinf f'(x)$. מצאנו סדרת פונקציות $f_n$ מדידות (הפרש מדידות הוא מדיד) שמתכנסת בכל $\R$ נקודתית ל־$f'$, ולכן $f'$ מדידה. 
	
	באופן כללי יותר גבול נקודתי של מדידות הוא מדיד. 
	
	בעצם, אנחנו מוכנים לוותר על התכנסות בחלק קטן מהנקודות (פרט לקבוצה במידה אפס). 
	
	\defi{במרחב $(\Og, \fc, \mu)$ מרחב מידה שלם. אומרים שתכונה מסוימת מתקיימת \textit{כמעט בכל מקום} אם היא מתקיימת לכל $x \in \Og$ פרט לקבוצה ממידה אפס. פורמלית $P \co \R \times X \to \{0, 1\}$ מקיימת $\pi_1\cl{P\op (0)}$ זניחה. }
	\rmark{בלעז, Almost Everywhere או AE. }
	\noti{$f, g \co \Og \to \R$ מסמנים $f \equiv g$ כב''מ (כמעט בכל מקום) אם $\{x \in \Og \co f(x) \neq g(x)\}$ זניחה, ון נקראות \textit{שקולות}. }
	\lem{שקילות כב''מ היא יסח שקילות. }
	
	\clm{תהא $f \co \Og \to \R$ מדידה, ונניח ש־$g \co \Og \to \R$ שווה ל־$f$ כב''מ. אזי גם $g$ מדידה. }\begin{proof}
		נסמן ב־$A$ את קבוצת הנקודות ש־$f, g$ לא מסכימות עליהן. $A$ זניחה. אזי לכל $t \in \R$: 
		\[ \{f < t\} \trio \{g < t\} \subset A \]
		ומשלמות $\mu((\{f < t\} \trio \{g < t\})) = 0$. וממדידות $\{f < t\}$ מקבלים ש־$\{g < t\}$ מדידה. 
	\end{proof}
	
	\textbf{דוגמה. }פונקציית דיריכלה $D(x) = \q_\Q$ שקולה לפונקציית האפס עד כדי קבוצה זניחה ולכן מדידה. 
	
	\defi{יהיו $f, f_1, f_2 \dots \co \Og \to \R\cup\{\pm\infty\}$. אומרים ש־$f_n \to f$ כב''מ אם כמעט לכל $x$ מתקיימת התכנסות נקודתית. }
	
	\textbf{דוגמאות. }
	\begin{itemize}
		\item $f_n(x) = x^{n}$ בקטע $[0, 1]$ מתכנס כב''מ ל־$0$. 
		\item עבור $x \in [0, 1]$, נסמן ב־$f_n(x)$ את מס' השביעיות בהצגה העשרונית של $x$. (לכאורה זה לא מוגדר היטב: למספרים מייצגו עשרוני סופי יש שתי הצגות, אבל המספרים מייצוג עשרוני צפופים מוכלים ב־$\Q$ והם קבוצה זניחה. אז $f_n(x)$ מוגדר היטב כב''מ). 
		
		אז: 
		\[ \lim_{n\to\infty} \frac{f_n(x)}{n} = \frac{1}{10} \]
		
		זה נקרא החוק החזק של המספרים הגדולים בהסתברות.  
	\end{itemize}
	
	\theo{גבול כב''מ של מדידות הוא מדיד. }\begin{proof}
		ה־$\limsup_n f_n(x)$ מדיד וכב''מ שווה לפונקציה הגבולית. 
	\end{proof}

	\subsection{התכנסות במידה}\ 
	
	בחדו''א 2א דיברנו על התכנסות במ''ש והתכנסות נקודתית. אז עכשיו יש לנו גרסאות חלשות יותר: 
	\[ \text{התכנסות במ''ש} \implies \text{התכנסות נקודתית} \implies \text{התכנסות כב''מ} \implies \text{התכנסות במידה} \]
	
	נגדיר את מושג ההתכנסות במידה. 
	
	\defi{נניח ש־$(\Og, \fc, \mu)$ מרחב מידה סופי. יהיו $f, f_1, f_2 \dots \co \Og \to \R$ מדידות. אז מסמנים $f_n \overset{\mu}{\to} f$ \textit{מתכנסות במידה} אם לכל $\eg > 0$: 
	\[ \mu\cl{\ccb{x \in \Og \co \sof{f_n(x)} - f(x)} > \eg}\toinf 0 \]}
	\rmark{בלעז, Convergence In Measurement}
	
	למה מרחב מידה סופי? בהתבוננות על: 
	\[ f(x) = \begin{cases}
		1 & x - \floor{x} \in [0, \frac{1}{n}] \\
		0 & \other
	\end{cases} \]
	אז $\{f =\frac{1}{2}\}$ ממידה אינסוף, ומההגדרה היא לא מתכנסת במידה (זה לא המצב הרצוי). אם מתעקשים על מרחב מידה אין־סופי, כדאי לדרוש שלכל קבוצה ממידה סופית, נדרוש שלכל $A \subset \Og$, אם $\mu(A) < \infty$ אז $f_n|_A \to f|_A$ במידה. 
	
	\textbf{דוגמה. }הגבנונים: בעבור הסדרה $I_1 =[0, 1/2], I_2 = [1/2, 1], \ I_3 = [0, 1/3], I_4 = [1/3, 2/3], I_5 = [2/3, 1], \ I_6 = [0, 1/4], I_7 = [1/4, 2/4]$ וכו'. נגדיר $f_n$ האינדיקטור על $I_n$. אז $f_n \toinf 0$ במידה. 
	
	צ''ל לכל $\eg > 0$ ש־$\{x \mid \sof{f_n(x) - 0} > \eg \} \doc A_n$ (בה''כ $0 < \eg < 1$) ממידה ששואפת לאפס כש־$n$ שואף לאינסוף. במקרה שלנו, משום שאורכי הקטעים שואפים ל־$0$, אז $\mu(A_n) = \eg_n$ ו־$\eg_n \to 0$. ולכן לכל $\eg \in (0, 1)$ הסדרה $\mu(A_n)$ שואפת ל־$0$ ולכן יש התכנסות למדיה בבירור. 
	
	האם יש התכנסות כמעט בכל מקום? לא – למעשה, אין התכנסות בשום נקודה $x$ ב־$[0, 1]$, כי הגבול העליון הוא הפונקציה $1$, כי לכל $x \in [0, 1]$ יש אינסוף ערכי $n$ שמכילים את $0$ ואינסוף ערכי $1$. 
	
	\defi{בהינתן מרחב מידה $(\Og, \fc, \mu)$, נאמר ש־$\Og' \subset \Og$ \textit{ממידה מלאה} אם $\mu(\Og \setminus \Og^{c}) = 0$ זניחה (בפרט $\Og' \in \fc$). }
	
	\theo{התכנסות כב''מ גוררת התכנסות במידה. }\begin{proof}
		נתונות $f, f_1, f_2 \dots \co \Og \to \R$ כך ש־$f_n(x) \toinf f(x)$ לכל $x \in \Og'$, כאשר $\Og' \subset \Og$ קבוצה ממידה מלאה. יהי $\eg > 0$, צ.ל. $\mu(\{x \in \Og \co \sof{f_n(x) - f(x)} > \eg\}) \toinf 0$. למעשה נוכיח משהו יותר חזק: נסמן: 
		\[ A_N = \ccb{x \in \Og \co \forall n \ge N \co \sof{f_n(x) - f(x)} \le \eg} \]
		מספיק להוכיח ש־$\mu(A_N^{c}) \to 0$. זאת כי: 
		\begin{align*}
			A_N^{c} = &\ \{x \in \Og \mid \exists n \ge \N\co \sof{f_N(x) - f(x)} > \eg\} \\
			\supset &\ \{x \in \Og \co \sof{f_N(x) - f(x) > \eg}\} \doc \ac_N
		\end{align*}
		לכן אם $\mu(A_N^{c}) \rrr{N \to \infty} 0$ אז גם $\ac_N \rrr{N \to \infty} 0$ (למעשה, $\mu(\ac_N) \tou 0$ במ''ש. זכרו זאת להמשך ההרצאה). נותר להראות שאכן $\mu(A_N^{c}) \rrr{N \to \infty} 0$. 
		
		לכל $x \in \Og'$ יודעים ש־$f_n(x) \to f(x)$. לכן קיים $N$ כך ש־$\forall n \ge N \co \sof{f_n(x) - f(x)} \le \eg$. כלומר, לכל $x \in \Og'$ קיים $N$ כך ש־$x \in A_N$. לכן: 
		\[ A \cod \bigcup_{n = 1}^{\infty}A_N \supset \Og' \]
		ובנוסף $A_1 \subset A_2 \subset \cdots$. משום ש־$\mu(A) = \mu(\Og)$ (כי $A \supset \Og'$. למעשה יש גם שימוש כאן בשלמות המרחב) ו־$A_N \nea A$, אז מרציפות המידה: 
		\[ \mu(A_N) \rrr{N \to \inft} \mu(A) = \mu(\Og) \]
		ובאופן שקול $\mu(A_N^{c}) \to 0$ כנדרש. 
	\end{proof}
	
	\begin{Theorem}[ריס]
		לכל סדרת פונקציות שמתכנסת במידה, יש ת''ס שמתכנסת כב''מ (לאותה פונקציה גבולית). 
	\end{Theorem}
	\rmark{באנגלית, Riesz. המון ריז בסקיבידי סיגמה אלגברה (הכותב \en{gen $\ag$}). }\begin{proof}
		נניח ש־$A_1, A_2 \dots \subset \Og$. נתיידד עם הקבוצות הבאות:
		\begin{gather*}
			\ac_G \cod \bigcap_{N = 1}^{\infty}\cl{\bigcup_{n \ge N}A_n} \\
			\ac_F \cod \bigcup_{N = 1}^{\infty}\cl{\bigcap_{n \ge N}A_n}
		\end{gather*}
		נבחין ש־$\ac_G$ היא קבוצת כל ה־$x$־ים ששייכים לאינסוף קבוצות, ו־$\ac_F$ קבוצת כל ה־$x$־ים שיייכים לכל הקבוצות פרט למספר סופי (המ''מ). נפנה להוכחת המשפט. 
		
		נתון ש־$f_n \toinf f$ במידה. עבור $\eg_k = \frac{1}{2^{k}}$ ידוע שהסדרה: 
		\[ \mu\cl{\ccb{x \in \Og \co \sof{f_n(x) - f(x)} > \eg_k}}\toinf 0 \]
		לכן, נקבל שבפרט קיים $n_k > 0$ כלשהו כך ש־: 
		\[ \forall n \ge n_k \co \mu\bigg(\underbrace{{\ccb{x \co \sof{f_n(x) - f(x)}> \frac{1}{2^{k}}}} }_{A_k}\bigg) < \frac{1}{2^{k}} \]
		אזי $\mu(A_k) < \frac{1}{2^{k}}$. הסדרה $A_k$ יוצרת $A_G$ לפי ההגדרה לעיל. נוכיח ש־: 
		\begin{enumerate}[(A)]
			\item $\mu(A_G) = 0$ (בורל־קנטלי)
			\item לכל $x \notin A_G$ מתקיים $f_{n_k}(x) \toinf f(x)$. 
		\end{enumerate}
		אחרי שנוכיח זאת, מצאנו ת''ס $n_k$ שעבורו $f_{n_k} \toinf f$ כב''מ. 
		\begin{enumerate}[(A)]
			\item לכל $n$ ידוע: 
			\[ \mu\cl{\bigcup_{k \ge n} A_k} \le \sum_{k = n}^{\infty}\mu(A_k) \le \sum_{k = n}^{\infty}\frac{1}{2^{k}} = \frac{1}{2^{k - 1}} \]
			ולכן: 
			\[ \mu(A_G) = \mu\Bigg(\bigcap_{n= 1}^{\infty}\underbrace{\csb{\bigcup_{k \ge n}A_k}}_{B_n\sea A_G}\Bigg) \sle \liminf_{n \to \infty}\mu\cl{\bigcup_{k \ge n}A_k} \le \lim_{n \to \infty}\frac{1}{2^{k - 1}} = 0 \]
			(למעשה $\smash{\sle}$ הוא שוויון במרחב מידה סופי, מרציפות המידה (רציפות לסדרה יורדת עובדת רק במרחב מידה סופי))
			\item נניח ש־$x \notin A_G$. אז קיים $N$ כך ש־$x \notin \bigcup_{k \ge n}A_k$, כלומר לכל $k \ge N$ מתקיים $x \notin A_k$. ואז $\forall k \ge N\co \sof{f_{n_k}(x) - f(x)} \le \frac{1}{2^{k}}$. \envendproof
		\end{enumerate}
	\end{proof}
	
	\textbf{סוד כמוס לפרה ולסוס. }התכנסות כב''מ היא לא מטרית ואפילו לא טופולוגית. למה הכוונה? אין שום טופולוגיה שמשרה התכנסות כב''מ. למה? בגלל משפט ריס. אילו הייתה טופולוגיה שמשרה את מושג ההתכנסות, אז היינו יכולים לקחת $f_n \to f$ שמתכנסת במידה ולא כב''מ, ואז הייתה איזושהי סביבה $U$ של $f$ כך שאינסוף פונקציות מחוץ ל־$U$ (שכן $f_n \not f$ כב''מ), נאמר $f_{n_1}, f_{n_2} \dots$. אך הן עדיין מתכנסות במידה, ולכן ישנה ת''ס $f_{n_{k_j}}$ שכן מתכנסת כב''מ אך זה לא מתאשר שכן $f_{n_{k_j}}$ לא בסביבה. 
	
	(התכנסות במ''ש מושרית ממטריקת הסופרמום והתכנסות נקודתית מטופולוגיית טיכונוף. התכנסות כב''מ היא ייצור מוזר בהקשר הזה). 
	
	\begin{Theorem}[Egorov]
		יהי $(\Og, \fc, \mu)$ מרחב מידה סופי. נניח ש־$f_n \toae f$ (כב''מ). יהי $\eg > 0$, אזי קיימת $A \subset \Og$ עם $\mu(\Og \setminus A) < \eg$ כך ש־: 
		\[ \sup_{x \in A}\sof{f_n(x) - f(x)} \toinf 0 \]
		במילים אחרות, $f_n|_A \tou f|_A$ במ''ש. 
	\end{Theorem}
	\textbf{דוגמה. }עבור $x_n(x) = x^{n}$ ב־$(0, 1)$, ישנה התכנסות לא במ''ש ל־$0$. אך אם מצטמצמים לקבוצה $(0, 1 - \eg)$ אז ההתכנסות במ''ש. 
	\rmark{זה לא עובר להתכנסות במידה. לדוגמה, הגבנון הנע. שם אין בשום נקודה התכנסות ובפרט לא במ''ש. }
	\begin{proof}
		תהא $\Og' \subset \Og$ קבוצה ממידה מלאה, כך ש־$f_n(x) \to f(x)$ לכל $x \in \Og'$.
		יהי $k \in \N$. נגדיר: 
		\[ A_N = \ccb{x \in \Og \mid \forall n \ge N \co \sof{f_n(x) - f(x)} < \frac{1}{2^{k}}} \]
		אזי $A_1 \subset A_2 \subset \cdots$, כלומר הן עולות ל־$A_N \nea A$ כלשהי. מטיעונים שכבר ראינו $\Og' \subset A$ ומרציפות $\mu$ נובע ש־$\mu(A_n) = \mu(A) = \mu(\Og') = \mu(\Og)$. 
		
		לכן קיים איזשהו $N_k$ כך ש־$E_k \cod A_{N_k}$ מקיימת $\mu(E_k^{c}) = \mu(A_{N_k}^{c}) = \frac{1}{2^{k}}$. 
		
		באופן כללי, ישנן קבוצות $E_1, E_2 \dots \subset \Og$ כך ש־: 
		\[ E_k = \ccb{x \in \Og \co \forall n \ge N_k \co \sof{f_n(x) - f(x)} < \frac{1}{2^{k}}} \]
		אזי $\mu(E_k^{c}) < \frac{1}{2^{k}}$. עתה נבחין שלכל $n$: 
		\[ \mu\cl{\Og \setminus \bigcap_{k \ge n} E_k^{c}} = \mu\cl{\bigcup_{k \ge n}E_k^{c}} \le \sum_{k \ge n}\mu(E_k^{c}) \le \sum_{k \ge n}\frac{1}{2^{k}} = \frac{2}{2^{n}}  \]
		ומכאן $\mu\cl{\bigcap_{k \ge n} E_k^{c}} = \mu(\Og) - \frac{2}{2^{n}}$. 
		
		יהי $\eg > 0$. משום ש־$\frac{2}{2^{n}}$ שואף ל־$0$ קיים $n$ כך ש־$B = \bigcap_{k \ge n}E_k$ מקיימת $\mu(B) = \mu(\Og) - \eg$. נותר להראות התכנסות במ''ש על $B$. 
		
		לכל $k$ מספיק גדול ($k \ge N_k$) מתקיים: 
		\[ \forall n \ge n_k \co \sup_{x \in A}\sof{f_n(x) - f(x)} \le \frac{1}{2^{k}} \]
		(זה שקול להתכנסות במ''ש) כי $x \in E_k$. 
	\end{proof}
	
	\begin{Theorem}[Luzin, ניסוח חלש]
		תהי $f \co \R\to \R$ מדידה. אז לכל $\eg  >0$ קיימת $A \subset \R$ סגורה כך ש־$\mu(\R \setminus A) < \eg$, כך ש־$f|_A$ רציפה. 
	\end{Theorem}
	הוכחה של משפט לוזים נראה בהמשך. 
	\rmark{יש הבדל בין ''$f$ רציפה ב־$A$`` ו־''$f|_A$ רציפה`` (הראשון אומר ש־$A$ קבוצת נקודות הרציפות של $f$, והשני אומר שהצמצום פונקציה רציפה). לדוגמה, $D(x)$ דיריכלה מקיימת $D|_\Q$ רציפה (היא פשוט פונקציית האפס), אך היא לא רציפה ב־$\Q$ (כי לכל $x \in \Q$, מתקיים ש־$x$ לא נקודת רציפות של $D(x)$). }
	\rmark{הן אגורוב והן לוזין נרדפו ע''י סטלין. אגורוב ''ממש חוסל`` (נאסר, שבת רעב בכלא, ומת), לוזין הוכרז כאויב במסכה של אזרח סובייטי. ב־2012 האקדמיה הרוסית למדעים ביטלה את ההחלטה. }

	\section[אינטגרל לבג]{\en{Lebesgue Integral}}
	יהי $(\Og, \fc, \mu)$ מרחב מידה. מה נצפה מאינטגרל? אנו כבר יודעים לעשות אינטגרציה לאינדנקטורים, ונצפה ש־: 
	\[ \int_\Og \q_A \dmu = \mu(A) \]
	זהו מושג האינטגרל הכי רחב שמשמש מתמטיקאים, לפחות לפונקציות חיוביות. הבנייה שלנו של אינטגרל לבג תהיה דו־שלבית: ראשית נגדיר אינטגרל לפונקציות פשוטות, ולאחר מכן לפונקציות כלליות.
	
	\subsection{אינטגרל של פונקציות פשוטות} \ 
	\defi{$f \co \Og \to \R$ היא \textit{פונקציה פשוטה} אם אוסף הערכים שהיא מקבלת הוא סופי או בן־מנייה, כלומר, הקבוצה $f(\Og) = \{f(x) \mid x \in \Og\}$ מקיימת $\sof{f(\Og)} \le \az$. }
	\rmark{השם ''פונקציה פשוטה`` לא מקובל במתמטיקה, הוא רק ישמש אותנו להגדיר את אינטגרל לבג. }
	\textbf{דוגמה. 
	\begin{enumerate}
		\item $\q_A$ האינדיקטורים פונקציה פשוטה
		\item ב־$\R$, $\sinx$ אינה פשוטה. 
	\end{enumerate}}

	איך נציג פונקציה פשוטה? כצירוף לינארי של אינדיקטורים: 
	\[ f = \sum_{n = 1}^{\infty}a_n1_{A_n} \]
	כאשר $A_1, A_2 \dots \subset \Og$ מדידות וזרות, ו־$a_1, a_2, \dots$ שונים זה מזה (פרט לאפסים). 
	
	הטור לעיל מתכנס נקודתית בכל מקום (משום שה־$A_n$־ים זרים). ברור ש־$A_n = f\op(a_n)$, ו־$f(\Og) = \{a_1, a_2 \dots\}$ (קיים מספור כזה כי $f(\Og)$ בת מנייה). 
	
	\clm{כל פונקציה פשוטה $f \co \Og \to \R$ אפשר לרשום כך: 
	\[ f = \sum_{k = 1}^{\wg} a_n \q_{A_n} \]
	כאשר $A_1, A_2 \dots$ מדידות, זרות ולא ריקות, $a_n$־ים שונים זה מזה, ו־$\wg = \N_0 \cup \{\infty\}$. }
	\prooftrivial
	\lem{ההצגה יחידה עד כדי: 
	\begin{itemize}
		\item שינוי סדר סכימה (די מתבקש, כי אין שום סדר קאנוני ל־$a_n$)
		\item אפשר להוסיף/להוציא איבר מהצורה $0 \cdot \q_A$ לטור. 
	\end{itemize}}
	\defi{יהי $(\Og, \fc, \mu)$ מרחב מידה, ו־$f = \sum_{n = 1}^{\wg}a_n\q_{A_n}$ פונקציה פשוטה בהצגה. אומרים ש־$f$ אינטגרבילית אם הטור $\sum_{a_n \neq 0}\sof{a_n}\mu(A_n) < \infty$. במקרה זה, נסמן: 
	\[ \int_\Og f \dmu = \int_{\Og}f(x) \dmu(x) = \sum_{a_n \neq 0}\sof{a_n}\mu(A_n) \]}
	\rmark{חשובה לנו מאוד התכנסות בהחלט, כדי שזה יהיה מוגדר היטב (הרי הייצוג כקומבינציה לינארית של אינדיקטורים לא תלוי בסדר סכימה). }
	\rmark{אם עבור $n$ מסוים, $a_n = 0$ ו־$\mu(A_n) = \infty$, אז אין זה משנה את ערך הטור. לשם כך נגדיר Abuse Of Notion לפיו אם $a_n =0 \land \mu(A_n) =\infty$, אז $a_n\mu(A_n) = 0$. זה יהפוך את הסימונים ליותר אלגנטיים. }
	\rmark{לעיתים נתיר פונקציות $f \co \Og \to \R \cup \{\pm\infty\}$ מדידות, ואז אם $a_n = \infty$ אך $\mu(A_n) =0$ זניחה, נגדיר שלא תהיה בעיה לדבר על אינטגרביליות שלה ונגדיר Abuse Of Notion של $a_n\mu(A_n) = 0$. }
	\rmark{באינטגרל לבג, אין ''כיוון`` לאינטגרציה. עושים אינטגרציה על קבוצה. לכן בתחתית האינטגרל כותבים את הקבוצה עליה עושים את האינטגרל, לא כותבים אינטגרל מ־$0$ עד $1$ (או משהו בסגנון). אין דבר כזה ''אינטגרל הפוך``. }
	
	\textbf{דוגמאות. }
